\section*{\color{AsianDarkBlue}I. HÀM CĂN BẢN TRONG C++}

\subsection*{\color{AsianBlue}1. Cấu trúc hàm tổng quát \& Hàm \texttt{void}}
Khi viết code, ta không nên dồn toàn bộ xử lý vào hàm \texttt{main()}. Việc tách nhỏ thành các hàm giúp code trong sáng, dễ đọc và dễ tìm lỗi.
\begin{lstlisting}[language=C++]
#include <iostream>
#include <string>
using namespace std;

// Ham void: Thuc hien cong viec ma khong tra ve gia tri
void say_hello(string name) {
    cout << "Xin chao ban: " << name << "!\n";
}

int main() {
    say_hello("Gia Huy"); // Goi ham
    say_hello("Minh Tri");
    return 0;
}
\end{lstlisting}

\subsection*{\color{AsianBlue}2. Hàm có giá trị trả về (\texttt{return})}
Hàm có kiểu trả về (như \texttt{long long}, \texttt{bool}, \texttt{double}) bắt buộc phải có lệnh \texttt{return} để trả kết quả tính toán về nơi gọi hàm.
\begin{lstlisting}[language=C++]
// Ham tinh binh phuong cua mot so nguyen
long long binh_phuong(long long x) {
    return x * x;
}

// Ham kiem tra mot so nguyen duong co phai so nguyen to hay khong
bool is_prime(long long n) {
    if (n < 2) return false;
    for (long long i = 2; i * i <= n; i++) {
        if (n % i == 0) return false;
    }
    return true;
}
\end{lstlisting}

\section*{\color{AsianDarkBlue}II. BẢN CHẤT TRUYỀN THAM TRỊ VS. THAM CHIẾU \texttt{\&}}

\subsection*{\color{AsianBlue}1. Phân biệt cơ chế hoạt động trên bộ nhớ RAM}
\begin{itemize}
    \setlength{\itemsep}{1pt}
    \item \textbf{Truyền tham trị (Pass-by-value):} Máy tính tạo ra một bản sao hoàn toàn mới trên ngăn xếp Stack. Mọi thay đổi bên trong hàm chỉ tác động lên bản sao, biến gốc bên ngoài \texttt{main()} giữ nguyên.
    \item \textbf{Truyền tham chiếu (Pass-by-reference \texttt{\&}):} Hàm nhận trực tiếp địa chỉ ô nhớ của biến gốc. Mọi thay đổi đối với tham số bên trong hàm sẽ làm thay đổi trực tiếp giá trị của biến gốc bên ngoài.
\end{itemize}

\begin{lstlisting}[language=C++]
#include <iostream>
using namespace std;

void tang_tham_tri(int x) {
    x = x + 1; // Chi tang ban sao, bien goc khong doi
}

void tang_tham_chieu(int &x) { // Co dau &
    x = x + 1; // Tac dong truc tiep vao bien goc ngoai main
}

int main() {
    int a = 10;
    tang_tham_tri(a);
    cout << "Sau tham tri: " << a << "\n";     // In ra: 10
    
    tang_tham_chieu(a);
    cout << "Sau tham chieu: " << a << "\n";   // In ra: 11
    return 0;
}
\end{lstlisting}

\subsection*{\color{AsianBlue}2. Ứng dụng kinh điển: Hàm tự hoán đổi 2 số (\texttt{my\_swap})}
Trong thuật toán sắp xếp hoặc xử lý mảng, ta cần hoán đổi giá trị hai biến. Bắt buộc phải dùng tham chiếu \texttt{\&}:
\begin{lstlisting}[language=C++]
void my_swap(long long &u, long long &v) {
    long long temp = u;
    u = v;
    v = temp;
}
\end{lstlisting}
