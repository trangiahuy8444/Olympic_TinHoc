\section*{\color{AsianDarkBlue}I. CHƯƠNG TRÌNH C++ ĐẦU TIÊN \& CÂU LỆNH \texttt{cout}}

\subsection*{\color{AsianBlue}1. Cấu trúc khung sườn của một file mã nguồn C++}
\begin{lstlisting}[language=C++]
#include <iostream>
using namespace std;

int main() {
    cout << "Chao mung cac ban den voi Asian School!\n";
    return 0;
}
\end{lstlisting}

\begin{itemize}
    \setlength{\itemsep}{1.5pt}
    \small
    \item \texttt{\#include <iostream>}: Khai báo thư viện vào/ra dữ liệu chuẩn (\textit{Input/Output Stream}).
    \item \texttt{using namespace std;}: Sử dụng không gian tên chuẩn để viết trực tiếp \texttt{cout, cin}.
    \item \texttt{int main() \{ ... \}}: Hàm chính của chương trình, nơi máy tính bắt đầu thực thi.
    \item \texttt{cout << "..." << "\textbackslash n";}: Lệnh in dữ liệu ra màn hình. Ký tự \texttt{\textbackslash n} dùng để xuống dòng.
    \item Dấu chấm phẩy \texttt{;} ở cuối mỗi lệnh đơn là \textbf{bắt buộc} trong C++.
\end{itemize}

\subsection*{\color{AsianBlue}2. In biểu thức tính toán trực tiếp}
\begin{lstlisting}[language=C++]
cout << "Tong 123 + 456 = " << 123 + 456 << "\n";
cout << "Tich 25 * 4 = " << 25 * 4 << "\n";
\end{lstlisting}
\textit{Quy tắc:} Muốn in chữ thì để trong dấu nháy kép \texttt{" "}; muốn máy tính tính ra kết quả số thì \textbf{không} dùng dấu nháy kép.

\section*{\color{AsianDarkBlue}III. NHẬP DỮ LIỆU VỚI \texttt{cin} \& CÁC TOÁN TỬ SỐ HỌC}

\subsection*{\color{AsianBlue}1. Nhập dữ liệu từ bàn phím với \texttt{cin >>}}
\begin{lstlisting}[language=C++]
int a, b;
cin >> a >> b; // Nguoi dung go 2 so cach nhau boi dau cach hoac Enter
cout << "Tong 2 so: " << a + b << "\n";
\end{lstlisting}

\subsection*{\color{AsianBlue}2. Phân biệt phép chia nguyên \texttt{/} và chia dư \texttt{\%}}
\begin{itemize}
    \setlength{\itemsep}{1.5pt}
    \small
    \item \textbf{Chia nguyên \texttt{a / b}:} Giữa 2 số nguyên sẽ trả về phần nguyên (ví dụ: \texttt{5 / 2 = 2}). Muốn ra \texttt{2.5} phải viết \texttt{5.0 / 2} hoặc \texttt{(double)a / b}.
    \item \textbf{Chia lấy dư \texttt{a \% b}:} Trả về số dư (ví dụ: \texttt{7 \% 3 = 1}; \texttt{2026 \% 10 = 6} giúp lấy chữ số hàng đơn vị).
\end{itemize}

\subsection*{\color{AsianBlue}3. Kỹ thuật hoán đổi 2 biến (Swap)}
\begin{lstlisting}[language=C++]
int a = 10, b = 20;
int temp = a; // Cat a vao bien tam
a = b;        // Chep b vao a
b = temp;     // Lay temp tra cho b -> a = 20, b = 10
\end{lstlisting}

\section*{\color{AsianDarkBlue}IV. BÀI TẬP TỰ LUYỆN TẠI NHÀ}
\begin{enumerate}
    \setlength{\itemsep}{2pt}
    \small
    \item \textbf{Bài 1.1:} Viết chương trình nhập vào 2 số nguyên dương $a, b$ ($1 \le a, b \le 10^6$). In ra chu vi và diện tích hình chữ nhật (chú ý kiểu \texttt{long long} cho diện tích).
    \item \textbf{Bài 1.2:} Nhập vào số giây $T$. Đổi sang \texttt{Giờ : Phút : Giây} bằng các phép toán \texttt{/} và \texttt{\%}.
    \item \textbf{Bài 1.3:} Nhập vào số nguyên dương $N$ có 3 chữ số (ví dụ: 358). Tính và in ra tổng 3 chữ số của $N$ ($3 + 5 + 8 = 16$).
\end{enumerate}
