\section*{\color{AsianDarkBlue}I. CÂU LỆNH RẼ NHÁNH \texttt{if}, \texttt{if-else}, \texttt{else if}}

\subsection*{\color{AsianBlue}1. Cú pháp và Toán tử so sánh}
\begin{lstlisting}[language=C++]
if (diem >= 8.0) {
    cout << "Hoc sinh Gioi\n";
} else if (diem >= 6.5) {
    cout << "Hoc sinh Kha\n";
} else {
    cout << "Can co gang them\n";
}
\end{lstlisting}

\begin{itemize}
    \setlength{\itemsep}{1.5pt}
    \small
    \item \textbf{Bẫy so sánh bằng:} Phải dùng 2 dấu bằng \texttt{==} (ví dụ: \texttt{if (a == b)}). Dấu \texttt{=} là lệnh gán!
    \item \textbf{Toán tử logic kết hợp:}
      \begin{itemize}
          \item \texttt{\&\&} (VÀ): Tất cả điều kiện con phải đúng (ví dụ: \texttt{if (a > 0 \&\& b > 0)}).
          \item \texttt{||} (HOẶC): Chỉ cần 1 điều kiện con đúng (ví dụ: \texttt{if (nam \% 400 == 0 || nam \% 4 == 0)}).
          \item \texttt{!} (PHỦ ĐỊNH): Đảo ngược giá trị logic (\texttt{!true = false}).
      \end{itemize}
\end{itemize}

\section*{\color{AsianDarkBlue}II. CẤU TRÚC VÒNG LẶP \texttt{for}}

\subsection*{\color{AsianBlue}1. Cú pháp cơ bản của vòng lặp \texttt{for}}
\begin{lstlisting}[language=C++]
// In cac so tu 1 den 10
for (int i = 1; i <= 10; i++) {
    cout << i << " ";
}
cout << "\n";
\end{lstlisting}
\textit{Thứ tự thực hiện:} (1) Khởi tạo \texttt{int i = 1} $\rightarrow$ (2) Kiểm tra \texttt{i <= 10} $\rightarrow$ (3) Thực hiện thân lệnh \texttt{cout} $\rightarrow$ (4) Tăng bước nhảy \texttt{i++} $\rightarrow$ Lặp lại (2).

\subsection*{\color{AsianBlue}2. Tính tổng dãy số $S = 1 + 2 + \dots + N$}
\begin{lstlisting}[language=C++]
long long tong = 0; // Bat buoc khoi tao tong = 0 truoc vong lap
for (int i = 1; i <= n; i++) {
    tong += i; // Tuong duong tong = tong + i
}
cout << "Tong = " << tong << "\n";
\end{lstlisting}

\section*{\color{AsianDarkBlue}III. VÒNG LẶP \texttt{while} \& KỸ THUẬT TÁCH CHỮ SỐ}

\subsection*{\color{AsianBlue}1. Vòng lặp \texttt{while}}
Dùng khi chưa biết trước số lần lặp, vòng lặp tiếp tục chừng nào điều kiện còn \texttt{true}:
\begin{lstlisting}[language=C++]
while (n > 0) {
    // Thuc hien khoi lenh ben trong
    // Can co lenh lam n giam dan ve 0 de tranh bi lap vo tan!
}
\end{lstlisting}

\subsection*{\color{AsianBlue}2. Kỹ thuật tách chữ số của số nguyên (Mổ xẻ số học)}
\begin{lstlisting}[language=C++]
// Tinh tong cac chu so va dem so luong chu so cua so N
long long n = 2026, temp = n;
int dem = 0, tong_cs = 0;
while (temp > 0) {
    int cs = temp % 10; // Lay chu so tan cung
    tong_cs += cs;      // Cong don vao tong chu so
    dem++;              // Dem so luong chu so
    temp /= 10;         // Bot chu so tan cung vua lay
}
// Ket qua: dem = 4, tong_cs = 10
\end{lstlisting}

\section*{\color{AsianDarkBlue}IV. BÀI TẬP TỰ LUYỆN TẠI NHÀ}
\begin{enumerate}
    \setlength{\itemsep}{2pt}
    \small
    \item \textbf{Bài 2.1:} Nhập số nguyên $N$. In ra \texttt{DUONG} nếu $N > 0$, in \texttt{AM} nếu $N < 0$, in \texttt{KHONG} nếu $N = 0$.
    \item \textbf{Bài 2.2:} Nhập 2 số $a, b$ ($a \le b$). Tính tổng các số chia hết cho 3 trong đoạn $[a, b]$.
    \item \textbf{Bài 2.3:} Nhập số nguyên dương $N$. Đếm xem $N$ có bao nhiêu ước số nguyên dương.
    \item \textbf{Bài 2.4:} Nhập số nguyên dương $N$. Kiểm tra xem $N$ có phải là số đối xứng (Palindrome) hay không (Ví dụ: 12321 là số đối xứng, 12345 không phải).
\end{enumerate}
