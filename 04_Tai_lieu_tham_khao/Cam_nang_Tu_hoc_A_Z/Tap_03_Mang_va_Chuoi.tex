\begin{itemize}

\section*{\color{AsianDarkBlue}I. MẢNG 1 CHIỀU CĂN BẢN}

\subsection*{\color{AsianBlue}1. Khai báo, Nhập và Xuất mảng}
\begin{lstlisting}[language=C++]
#include <iostream>
using namespace std;

int main() {
    int n;
    cin >> n;
    int a[105]; // Khai bao mang du dung cho n phan tu

    for (int i = 0; i < n; i++) cin >> a[i]; // Nhap mang tu ban phim

    for (int i = 0; i < n; i++) cout << a[i] << " "; // Xuat mang
    cout << "\n";
    return 0;
}
\end{lstlisting}

\subsection*{\color{AsianBlue}2. Tìm Max, Min và Tính tổng mảng}
\begin{lstlisting}[language=C++]
int max_val = a[0]; // LUON khoi tao bang a[0], tuyet doi khong khoi tao = 0
long long tong = 0;

for (int i = 0; i < n; i++) {
    if (a[i] > max_val) max_val = a[i];
    tong += a[i];
}
cout << "Max = " << max_val << ", Tong = " << tong << "\n";
\end{lstlisting}

\section*{\color{AsianDarkBlue}II. CHUỖI KÝ TỰ \texttt{std::string} CĂN BẢN}

\subsection*{\color{AsianBlue}1. Khai báo chuỗi \& Nhập xuất văn bản}
\begin{lstlisting}[language=C++]
#include <iostream>
#include <string>
using namespace std;

int main() {
    string s1, s2;
    cin >> s1; // Chi doc 1 tu (dung lai o dau cach hoac Enter)
    
    cin.ignore(); // Xoa dau Enter thua truoc khi dung getline
    getline(cin, s2); // Doc toan bo mot dong van ban co khoang trang

    cout << "Do dai chuoi s2: " << s2.length() << "\n";
    cout << "Ky tu dau tien: " << s2[0] << "\n";
    return 0;
}
\end{lstlisting}

\subsection*{\color{AsianBlue}2. Duyệt từng ký tự và đếm ký tự trong chuỗi}
\begin{lstlisting}[language=C++]
// Dem so luong chu cai in hoa trong chuoi
int dem_hoa = 0;
for (int i = 0; i < s.length(); i++) {
    if (s[i] >= 'A' && s[i] <= 'Z') {
        dem_hoa++;
    }
}
cout << "So chu hoa: " << dem_hoa << "\n";
\end{lstlisting}

\section*{\color{AsianDarkBlue}III. BÀI TẬP TỰ LUYỆN TẠI NHÀ}
\begin{enumerate}
    \setlength{\itemsep}{2pt}
    \small
    \item \textbf{Bài 3.1:} Nhập mảng $N$ số nguyên. Tính và in ra trung bình cộng của các số dương trong mảng.
    \item \textbf{Bài 3.2:} Nhập mảng $N$ số nguyên. Đếm xem có bao nhiêu phần tử chia hết cho 5 nhưng không chia hết cho 2.
    \item \textbf{Bài 3.3:} Nhập một chuỗi văn bản $S$. Đếm xem trong chuỗi có bao nhiêu ký tự chữ số (\texttt{'0'} đến \texttt{'9'}).
    \item \textbf{Bài 3.4:} Nhập một chuỗi $S$. Hãy in ra chuỗi đó theo thứ tự đảo ngược từ ký tự cuối về ký tự đầu.
\end{enumerate}
