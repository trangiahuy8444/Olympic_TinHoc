\section*{\color{AsianDarkBlue}2. Tra cứu Cấu trúc Dữ liệu C++ STL Cốt lõi}

\subsection*{\color{AsianBlue}2.1. Mảng động: \texttt{std::vector}}
\begin{lstlisting}[language=C++]
#include <vector>
#include <algorithm>

vector<int> a;
a.push_back(10);                // Them phan tu vao cuoi
int x = a.back(); a.pop_back(); // Lay va xoa phan tu cuoi
int n = a.size();               // Kich thuoc mang
sort(a.begin(), a.end());       // Sap xep tang dan
reverse(a.begin(), a.end());    // Dao nguoc mang
int first = a[0], last = a.back();
\end{lstlisting}

\subsection*{\color{AsianBlue}2.2. Tập hợp: \texttt{std::set}}
\begin{lstlisting}[language=C++]
#include <set>

set<int> s; // Tu dong sap xep tang dan, loai bo trung lap
s.insert(5);                 // Them phan tu
if (s.count(5)) s.erase(5);  // Kiem tra va xoa phan tu
int n = s.size();

for (int x : s) cout << x << " "; // Duyet tap hop
\end{lstlisting}

\subsection*{\color{AsianBlue}2.3. Bảng ánh xạ / Đếm tần số: \texttt{std::map}}
\begin{lstlisting}[language=C++]
#include <map>

map<string, int> d;
d["apple"] = 3;
if (d.count("apple")) d["apple"]++;

// Dem tan so cac phan tu mang trong O(N log N)
map<int, int> freq;
for (int x : arr) freq[x]++;
\end{lstlisting}

\subsection*{\color{AsianBlue}2.4. Cặp giá trị: \texttt{std::pair}}
\begin{lstlisting}[language=C++]
#include <utility>
#include <vector>
#include <algorithm>

pair<int, int> p = {10, 20};
int x = p.first, y = p.second;

vector<pair<int, int>> points = {{1, 2}, {0, 5}, {1, 1}};
sort(points.begin(), points.end()); // Tu dong so sanh first roi den second!
\end{lstlisting}

\section*{\color{AsianDarkBlue}3. Bảng quy tắc độ phức tạp và giới hạn $N$ (1.0 giây)}
\begin{tcolorbox}[colback=red!5!white,colframe=red!75!black,title=\textbf{QUY TẮC PHÂN TÍCH THUẬT TOÁN THEO SUBTASK}]
\begin{itemize}
    \item $N \le 20$: Thuật toán vét cạn, quay lui $O(2^N)$ hoặc $O(N!)$.
    \item $N \le 500$: Thuật toán $O(N^3)$ (Floyd, 3 vòng for lồng nhau).
    \item $N \le 5000$: Thuật toán $O(N^2)$ (Quy hoạch động 2 chiều, 2 vòng for).
    \item $N \le 10^5 \dots 10^6$: Thuật toán $O(N \log N)$ hoặc $O(N)$ (Sắp xếp, Two Pointers, Prefix Sum, Chặt nhị phân).
\end{itemize}
\end{tcolorbox}
