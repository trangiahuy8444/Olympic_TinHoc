\section*{\color{AsianDarkBlue}\large II. Hướng dẫn khai thác 3 kho tài nguyên trực tuyến}

\subsection*{\color{AsianBlue}\normalsize 1. VisuAlgo (\url{https://visualgo.net/}) -- Trực quan hóa Thuật toán}
\begin{itemize}
    \setlength{\itemsep}{2pt}
    \item \textbf{Mục đích:} Giúp học sinh "nhìn thấy" cách dữ liệu di chuyển và thuật toán vận hành từng bước qua hình ảnh động trực quan. Website có hỗ trợ tiếng Việt ở menu góc phải.
    \item \textbf{5 Module cốt lõi cần thực hành:}
    \begin{enumerate}
        \setlength{\itemsep}{1pt}
        \item \textbf{Sorting (Sắp xếp):} Quan sát cơ chế so sánh và tráo đổi của \textit{Bubble Sort}, \textit{Merge Sort}, và \textit{Quick Sort}.
        \item \textbf{BST (Binary Search Tree):} Tự gõ các số để xem cây tự cân bằng (\texttt{std::set}) chèn phần tử và tìm kiếm (\texttt{lower\_bound}) theo chiều cao $O(\log N)$.
        \item \textbf{Graph Traversal (Duyệt đồ thị):} Xem đồ họa so sánh \textit{BFS} (loang theo lớp sóng bằng Queue) và \textit{DFS} (đi sâu kịch đường bằng Stack rồi quay lui).
        \item \textbf{SSSP (Đường đi ngắn nhất):} Xem thuật toán \textit{Dijkstra} lan truyền và cập nhật khoảng cách bằng hàng đợi ưu tiên.
        \item \textbf{MST (Cây khung nhỏ nhất):} Xem cách thuật toán \textit{Kruskal} sắp xếp các cạnh và dùng cấu trúc DSU để tránh tạo chu trình.
    \end{enumerate}
\end{itemize}

\subsection*{\color{AsianBlue}\normalsize 2. VNOI Wiki (\url{https://wiki.vnoi.info/}) -- Bách khoa toàn thư Thuật toán Việt Nam}
\begin{itemize}
    \setlength{\itemsep}{2pt}
    \item \textbf{Mục đích:} Cung cấp lời giảng lý thuyết chuẩn mực bằng tiếng Việt, code mẫu C++ chuẩn mực và hệ thống nộp bài luyện tập trực tiếp trên \textbf{VNOJ} (\url{https://oj.vnoi.info/}).
    \item \textbf{Phương pháp học:} Khi đọc sách tiếng Anh của Antti Laaksonen gặp khái niệm mới, các em mở ngay VNOI Wiki để đọc bài viết tương ứng bằng tiếng Việt. Cuối mỗi bài viết luôn có link bài tập thực hành trên VNOJ để nộp code chấm tự động.
    \item \textbf{Các chuyên đề đọc hàng tuần:}
    \begin{itemize}
        \setlength{\itemsep}{1pt}
        \item Thư viện C++ STL căn bản: \url{https://wiki.vnoi.info/algo/basic/c++-stl}
        \item Kỹ thuật Hai con trỏ: \url{https://wiki.vnoi.info/algo/basic/two-pointers}
        \item Mảng cộng dồn Prefix Sum: \url{https://wiki.vnoi.info/algo/basic/prefix-sum}
        \item Quy hoạch động nhập môn: \url{https://wiki.vnoi.info/algo/dp/basic-dp}
        \item Thuật toán trên đồ thị: \url{https://wiki.vnoi.info/algo/graph/dfs-bfs}
    \end{itemize}
\end{itemize}

\subsection*{\color{AsianBlue}\normalsize 3. USACO Guide (\url{https://usaco.guide/}) -- Giáo trình Luyện thi Olympic Hoa Kỳ}
\begin{itemize}
    \setlength{\itemsep}{2pt}
    \item \textbf{Mục đích:} Nền tảng phân bậc bài tập chuẩn hóa số 1 thế giới hiện nay cho học sinh phổ thông.
    \item \textbf{Cách học từng bước:}
    \begin{enumerate}
        \setlength{\itemsep}{1pt}
        \item Đăng nhập miễn phí bằng tài khoản Google $\to$ Chọn ngôn ngữ mặc định là \textbf{C++17}.
        \item \textbf{Giai đoạn 1 (2 tháng đầu):} Làm chủ toàn bộ các bài tập mức \textbf{Bronze} (Hạng Đồng).
        \item \textbf{Giai đoạn 2 (Sau tháng 11):} Chuyển sang cày mức \textbf{Silver} (Hạng Bạc) -- đây chính là độ khó tương đương với các bài 2 và bài 3 trong đề thi Olympic 11 TP.HCM.
    \end{enumerate}
\end{itemize}
