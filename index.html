NGHE NHẠC (NGHENHAC.CPP)

Dạng bài: Hai Con Trỏ (Two Pointers) Cửa Sổ Trượt

Tóm tắt đề: An có $K$ phút rảnh và danh sách $N$ bài hát, bài thứ $i$ dài $m_i$ phút ($1 \le N \le 10^5, 1 \le K \le 10^9, 1 \le m_i \le 10^4$). Hãy tìm một dãy liên tiếp các bài hát sao cho số lượng bài hát nhiều nhất và tổng thời lượng $\le K$.

Subtask 1 (60% số điểm - 3.6đ): $N \le 10^3$. Duyệt 2 vòng for $O(N^2)$ vẫn kịp giờ.
Subtask 2 (40% số điểm - 2.4đ): $N \le 10^5$. Bắt buộc dùng Hai con trỏ đạt $O(N)$ thời gian, $O(1)$ bộ nhớ!
NGHENHAC.CPP O(N) Thời gian • O(1) Bộ nhớ 
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("NGHENHAC.INP", "r", stdin);
    freopen("NGHENHAC.OUT", "w", stdout);
    #endif

    int n;
    long long k;
    if (!(cin >> n >> k)) return 0;

    vector<long long> m(n);
    for (int i = 0; i < n; i++) cin >> m[i];

    int left = 0, ans = 0;
    long long current_sum = 0;

    // Kỹ thuật Hai Con Trỏ (Sliding Window) chạy trong O(N)
    for (int right = 0; right < n; right++) {
        current_sum += m[right];
        while (current_sum > k && left <= right) {
            current_sum -= m[left];
            left++;
        }
        ans = max(ans, right - left + 1);
    }

    cout << ans << "
";
    return 0;
}

HỌC TOÁN (HOCTOAN.CPP)

Dạng bài: Số Học (Number Theory) GCD & LCM

Tóm tắt đề: Cho hai số $a, b$ ($1 \le a < b \le 10^4$). Tìm hai số $x, y$ sao cho $\gcd(x, y) = a$, $\text{lcm}(x, y) = b$ và tổng $x + y$ nhỏ nhất.

Phân tích toán học: Ta có $x \cdot y = \gcd(x, y) \cdot \text{lcm}(x, y) = a \cdot b$. Đặt $x = a \cdot u$ và $y = a \cdot v$ với $\gcd(u, v) = 1$. Khi đó $u \cdot v = b / a$. Duyệt $u$ từ $\lfloor\sqrt{b/a}\rfloor$ giảm dần về 1, độ phức tạp chỉ mất $O(\sqrt{b/a})$!

HOCTOAN.CPP O(log(min(A,B))) Euclid 
#include <iostream>
#include <numeric>
#include <cmath>

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("HOCTOAN.INP", "r", stdin);
    freopen("HOCTOAN.OUT", "w", stdout);
    #endif

    long long a, b;
    if (!(cin >> a >> b)) return 0;

    if (b % a != 0) {
        cout << -1 << "
";
        return 0;
    }

    long long P = b / a;
    long long best_u = -1, best_v = -1;

    // Duyệt từ căn bậc 2 giảm dần để tổng x + y nhỏ nhất
    for (long long u = sqrt(P); u >= 1; u--) {
        if (P % u == 0) {
            long long v = P / u;
            if (std::gcd(u, v) == 1) {
                best_u = u;
                best_v = v;
                break;
            }
        }
    }

    long long x = a * best_u;
    long long y = a * best_v;
    if (x > y) swap(x, y);

    cout << (x + y) << "
";
    cout << x << " " << y << "
";
    return 0;
}

ROBOT (ROBOT.CPP)

Dạng bài: Mô Phỏng Không Gian Lớn std::set & pair

Tóm tắt đề: Bảng $M \times N$ ($2 \le M, N \le 10^5$). Robot 1 đi từ $(1, 1) \to (M, N)$ theo lệnh `D` (xuống) và `R` (phải). Robot 2 đi từ $(M, N) \to (1, 1)$ theo `U` (lên) và `L` (trái). Đếm số ô mà cả 2 robot cùng đi qua.

Bẫy kinh điển: Kích thước bảng $10^5 \times 10^5 = 10^{10}$ ô $\implies$ Không thể tạo mảng 2 chiều! Nhưng mỗi robot chỉ đi tối đa $(M-1) + (N-1) \le 2 \cdot 10^5$ bước. Dùng std::set<pair<int, int>> lưu các tọa độ robot 1 đi qua trong $O((M+N) \log(M+N))$, chạy chỉ mất 0.15s!

ROBOT.CPP O(M log K) std::set Tọa Độ 
#include <iostream>
#include <string>
#include <set>

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("ROBOT.INP", "r", stdin);
    freopen("ROBOT.OUT", "w", stdout);
    #endif

    int m, n;
    if (!(cin >> m >> n)) return 0;

    string s1, s2;
    cin >> s1 >> s2;

    // Lưu vết đường đi Robot 1
    set<pair<int, int>> path1;
    int r = 1, c = 1;
    path1.insert({r, c});

    for (char cmd : s1) {
        if (cmd == 'D') r++;
        else if (cmd == 'R') c++;
        path1.insert({r, c});
    }

    // Mô phỏng Robot 2 và đếm số ô trùng nhau
    set<pair<int, int>> common_cells;
    r = m; c = n;
    if (path1.count({r, c})) common_cells.insert({r, c});

    for (char cmd : s2) {
        if (cmd == 'U') r--;
        else if (cmd == 'L') c--;
        if (path1.count({r, c})) {
            common_cells.insert({r, c});
        }
    }

    cout << common_cells.size() << "
";
    return 0;
}

DI TRUYỀN (DNA.CPP)

Xử Lý Xâu & Bảng Vị Trí

Tóm tắt đề: $N \le 10$ chuỗi DNA độ dài $\le 10^5$. Một chuỗi là bất thường nếu: (1) Ít hơn 3 loại ký tự, (2) Có ký tự chỉ xuất hiện 1 lần, hoặc (3) Khoảng cách giữa 2 ký tự cùng loại liên tiếp $> K$. Tạo chuỗi nhị phân và đổi sang hệ thập phân.

Thuật toán: Duyệt chuỗi trong $O(|S|)$, lưu vị trí gần nhất `last_pos[c]`. Đổi nhị phân sang thập phân bằng công thức $ans = ans \times 2 + bit$. Độ phức tạp: $O(N \cdot |S|)$ cực nhanh!

DNA.CPP O(N · |S|) Xử Lý Xâu Tối Ưu 
#include <iostream>
#include <string>
#include <map>

using namespace std;

bool is_abnormal(const string &s, int k) {
    map<char, int> freq, last_pos;
    for (int i = 0; i < (int)s.size(); i++) {
        char c = s[i];
        freq[c]++;
        if (last_pos.count(c) && (i - last_pos[c]) > k) return true;
        last_pos[c] = i;
    }
    if (freq.size() < 3) return true;
    for (auto &p : freq) if (p.second == 1) return true;
    return false;
}

int main() {
    ios_base::sync_with_stdio(false); cin.tie(NULL);
    #ifndef ONLINE_JUDGE
    freopen("DNA.INP", "r", stdin); freopen("DNA.OUT", "w", stdout);
    #endif
    int n, k; cin >> n >> k;
    long long decimal_ans = 0;
    for (int i = 0; i < n; i++) {
        string s; cin >> s;
        decimal_ans = decimal_ans * 2 + (is_abnormal(s, k) ? 1 : 0);
    }
    cout << decimal_ans << "
";
    return 0;
}

THU HOẠCH (THUHOACH.CPP)

Quy Hoạch Động (DP) + Prefix Sum

Tóm tắt đề: $N \le 5000$ thửa ruộng. Chọn $K$ máy gặt, mỗi máy gặt $L$ thửa liên tiếp ($K \times L \le N$) không đè lên nhau để tổng giá trị lớn nhất.

Thuật toán: Mảng cộng dồn $P[i]$ tính tổng đoạn $L$. Bảng phương án: $dp[i][j] = \max(dp[i-1][j], dp[i-L][j-1] + val[i])$ với $val[i] = P[i] - P[i-L]$. Độ phức tạp: $O(N \cdot K) \le 2.5 \cdot 10^7$ phép tính, chạy trong 0.08s!

using namespace std;

const long long INF = 1e18;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("THUHOACH.INP", "r", stdin);
    freopen("THUHOACH.OUT", "w", stdout);
    #endif

    int n, k, l;
    if (!(cin >> n >> k >> l)) return 0;

    vector a(n + 1), pref(n + 1, 0);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        pref[i] = pref[i - 1] + a[i];
    }

    // dp[i][j]: Tổng lớn nhất khi xét i thửa ruộng đầu và chọn j đoạn độ dài L
    vector> dp(n + 1, vector(k + 1, -INF));
    for (int i = 0; i <= n; i++) dp[i][0] = 0;

    for (int i = l; i <= n; i++) {
        long long sum_L = pref[i] - pref[i - l];
        for (int j = 1; j <= k; j++) {
            dp[i][j] = dp[i - 1][j]; // Không gặt thửa i
            if (dp[i - l][j - 1] != -INF) {
                dp[i][j] = max(dp[i][j], dp[i - l][j - 1] + sum_L);
            }
        }
    }

    cout << dp[n][k] << "
";
    return 0;
}

BỘ SƯU TẬP (SUUTAP.CPP)

Hai Con Trỏ + std::map

Tóm tắt đề: Dãy $N \le 10^5$ viên đá có mã màu $C_i \le 10^9$. Tìm đoạn liên tiếp ngắn nhất chứa ít nhất $K$ màu khác nhau.

Thuật toán: Vì $C_i \le 10^9$, dùng map<int, int> freq đếm số lượng màu trong cửa sổ. Tăng `right` nạp màu. Khi `freq.size() >= k`, cập nhật `min_len` và tăng `left` để thu hẹp. Độ phức tạp $O(N \log K)$!

SUUTAP.CPP O(N log K) Hai Con Trỏ + map 
#include 
#include 
#include 
#include 

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("SUUTAP.INP", "r", stdin);
    freopen("SUUTAP.OUT", "w", stdout);
    #endif

    int n, k;
    if (!(cin >> n >> k)) return 0;

    vector c(n);
    for (int i = 0; i < n; i++) cin >> c[i];

    map freq;
    int left = 0, min_len = 1e9;

    for (int right = 0; right < n; right++) {
        freq[c[right]]++;
        while ((int)freq.size() >= k) {
            min_len = min(min_len, right - left + 1);
            freq[c[left]]--;
            if (freq[c[left]] == 0) freq.erase(c[left]);
            left++;
        }
    }

    if (min_len > n) cout << -1 << "\n";
    else cout << min_len << "\n";
    return 0;
}

BÀI TOÁN SLTPLT
Đồ Thị: Đếm Số Thành Phần Liên Thông (BFS/DFS)

Biểu diễn đồ thị bằng danh sách kề `vector adj[]`, duyệt BFS tìm vùng liên thông và kỹ thuật mảng đánh dấu `visited[]`.

#include 

using namespace std;

const long long INF = 1e18;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("LOCO.INP", "r", stdin);
    freopen("LOCO.OUT", "w", stdout);
    #endif

    int m, n;
    if (!(cin >> m >> n)) return 0;

    vector> a(m + 1, vector(n + 1));
    for (int i = 1; i <= m; i++)
        for (int j = 1; j <= n; j++) cin >> a[i][j];

    // dp[i][j]: Chi phí tối ưu đến ô (i, j)
    vector> dp(m + 1, vector(n + 1, INF));
    dp[1][1] = a[1][1];

    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (i == 1 && j == 1) continue;
            long long prev_min = INF;
            if (i > 1) prev_min = min(prev_min, dp[i - 1][j]);
            if (j > 1) prev_min = min(prev_min, dp[i][j - 1]);
            if (prev_min != INF) dp[i][j] = prev_min + a[i][j];
        }
    }

    cout << dp[m][n] << "\n";
    return 0;
}

#include 
#include 
#include 

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("SLTPLT.INP", "r", stdin);
    freopen("SLTPLT.OUT", "w", stdout);
    #endif

    int n, m;
    if (!(cin >> n >> m)) return 0;

    vector> adj(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v; cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    vector visited(n + 1, false);
    int components = 0;

    for (int i = 1; i <= n; i++) {
        if (!visited[i]) {
            components++;
            queue q;
            q.push(i);
            visited[i] = true;
            while (!q.empty()) {
                int u = q.front(); q.pop();
                for (int v : adj[u]) {
                    if (!visited[v]) {
                        visited[v] = true;
                        q.push(v);
                    }
                }
            }
        }
    }

    cout << components << "\n";
    return 0;
}

3 Kỹ Thuật I/O Sống Còn Trong Phòng Thi

Chống Trượt TLE
1. ios_base::sync_with_stdio(false);

Mặc định C++ đồng bộ luồng cin/cout với thư viện C (scanf/printf) để an toàn. Lệnh này ngắt đồng bộ, giúp cin/cout đạt tốc độ ngang ngửa thậm chí nhanh hơn scanf/printf!

2. cin.tie(NULL);

Ngắt liên kết tự động xả bộ đệm (flush) của cout trước mỗi lệnh cin. Trong các bài toán có nhiều thao tác nhập xuất xen kẽ, lệnh này giúp tiết kiệm tới 50% thời gian thực thi.

3. #ifndef ONLINE_JUDGE và freopen

Khi chấm trên các hệ thống chấm tự động (VNOJ, LQDOJ, Codeforces), macro ONLINE_JUDGE tự động kích hoạt giúp đọc từ Standard I/O. Khi thi offline của Sở, code tự động chuyển sang đọc ghi tệp .INP/.OUT chuẩn xác!

Bảng Vàng Quy Đổi Giới Hạn $N \implies$ Thuật Toán Tối Ưu

Quy chuẩn máy chấm thi Olympic: 1.0 giây $\approx 10^8$ phép tính. Luôn nhìn vào ràng buộc $N$ trong đề để chọn đúng thuật toán!

Đọc VNOI Wiki: Độ Phức Tạp
Giới hạn $N$ trong đề bài Độ phức tạp $O(...)$ Thuật toán bắt buộc áp dụng Ví dụ bài toán kinh điển Chiến lược điểm thi
$N \le 10 - 12$ $O(N!)$ hoặc $O(N^2 \cdot 2^N)$ Duyệt mọi hoán vị (std::next_permutation), Quy hoạch động trạng thái (Bitmask DP). Người du lịch (TSP), Phân việc cho $N$ người, Hoán vị chữ số. Ăn trọn $100\%$
$N \le 20 - 25$ $O(2^N)$ Duyệt quay lui (Backtracking), Vét cạn có nhánh cận. Thử tất cả tập con. Cái túi (Knapsack Subtask 1), Xếp $N$ quân hậu, Chia kẹo 2 phần bằng nhau. Vét Subtask 1 (30-40%)
$N \le 400$ $O(N^3)$ 3 vòng for lồng nhau, Nhân ma trận cơ bản, Floyd-Warshall tìm đường đi ngắn nhất. Nhân 2 ma trận $N \times N$, Đường đi ngắn nhất mọi cặp đỉnh (Floyd), Đếm tam giác đồ thị. Ăn Subtask 2 (60%)
$N \le 2\,000 - 5\,000$ $O(N^2)$ 2 vòng for lồng nhau, Quy hoạch động 2 chiều, Duyệt mọi cặp $(i, j)$. Dãy con tăng dài nhất (LIS DP cơ bản), Xâu con chung dài nhất (LCS), Đếm cặp nghịch thế. Ăn Subtask 2 (60%)
$N \le 10^5 - 2 \cdot 10^5$ $O(N \log N)$ Sắp xếp (std::sort), Tìm kiếm nhị phân, Hai con trỏ, Set/Map, Priority Queue. Đếm cặp có tổng bằng $K$ (Two Pointers/BS), Tìm giá trị nhỏ nhất thỏa điều kiện (Chặt nhị phân kết quả), Segment Tree. Ăn Trọn 100% Bài 1, 2
$N \le 10^6 - 10^7$ $O(N)$ hoặc $O(N \log \log N)$ Mảng đếm tần số (Direct Addressing Table), Mảng cộng dồn (Prefix Sum), Sàng Eratosthenes. Đếm số lần xuất hiện của các phần tử, Tính tổng đoạn con $Q$ truy vấn $O(1)$, Sàng lọc số nguyên tố $\le N$. Ăn Trọn 100% Bài 1, 2
$N \ge 10^9 - 10^{18}$ $O(\log N)$ hoặc $O(1)$ Công thức toán học đóng, Thuật toán Euclid GCD, Lũy thừa nhị phân $A^B \pmod M$. Ước chung lớn nhất $\gcd(A, B)$, Lũy thừa nhanh $A^B \pmod{10^9+7}$, Tính số Fibonacci thứ $10^{18}$ (Nhân ma trận). Giải Nhất Tuyệt Đối
3 Tình Huống Thực Chiến: Cách Nhìn Giới Hạn $N$ Để Đoán Ngay Thuật Toán Trong 3 Giây Bí kíp phòng thi Olympic
Ví dụ 1: Đếm cặp số có tổng bằng $K$ Two Sum

Đề bài: "Cho mảng $A$ có $N$ số nguyên. Hãy đếm số cặp $(i, j)$ ($i < j$) sao cho $A_i + A_j = K$."

❌ Nếu $N = 10^5$: Duyệt 2 vòng for lồng nhau $O(N^2) = 10^{10} \implies$ TLE (100 giây) ăn 0 điểm!
✅ Giải pháp đúng: Sort mảng + Kỹ thuật Hai con trỏ $O(N \log N) \approx 1.7 \cdot 10^6 \implies$ Chạy 0.02s ăn trọn 100%!
Ví dụ 2: Truy vấn tổng đoạn $[L, R]$ Prefix Sum

Đề bài: "Cho mảng $N$ phần tử và $Q$ câu hỏi. Mỗi câu hỏi tính tổng các số từ vị trí $L$ đến $R$ ($N, Q \le 10^5$)."

❌ Mỗi câu hỏi chạy vòng for từ $L$ đến $R$: $O(Q \times N) = 10^{10} \implies$ TLE mất 100s ăn 0 điểm!
✅ Mảng cộng dồn (Prefix Sum): Tiền xử lý $O(N)$, trả lời mỗi câu $O(1) \implies$ Chạy 0.03s ăn trọn 100%!
Ví dụ 3: Giới hạn số khổng lồ $10^{18}$ Math O(1)

Đề bài: "Tính tổng các số tự nhiên từ $1$ đến $N$ với giới hạn $N \le 10^{18}$."

❌ Chạy vòng lặp for (long long i=1; i<=N; ++i): $10^{18}$ bước $\implies$ Mất hơn 300 năm!
✅ Dùng công thức Gauss $S = \frac{N \times (N + 1)}{2}$ với $O(1) \implies$ Chạy 0.000001s ăn trọn 100%!
Quy tắc ước lượng bộ nhớ RAM (Memory Limit 256MB/512MB):
10^7 phần tử int $\approx 40\text{MB}$ 10^7 phần tử long long $\approx 80\text{MB}$ 10^7 phần tử bool $\approx 10\text{MB}$

5 Bẫy Tử Thần Phòng Thi (Mất Điểm Oan & Cách Tránh)

Hơn 40% Thí Sinh Dính Bẫy
1. Nhân 2 số int gán vào long long

Code: long long p = a * b; (với $a, b$ là int). Phép nhân được tính trên thanh ghi 32-bit của CPU trước, tràn thành số âm rồi mới gán vào p!

Khắc phục: 1LL * a * b hoặc ép kiểu (long long)a * b.
2. Phép chia nguyên làm tròn lên

Tính $\lceil a / b \rceil$ (xếp người vào xe/phòng). Dùng ceil((double)a / b) dễ bị sai số dấu phẩy động khi $a, b \ge 10^{14}$!

Khắc phục: (a + b - 1) / b với $a, b > 0$.
3. Trôi dòng getline sau cin >> n

Sau khi đọc số cin >> n;, ký tự xuống dòng '\n' vẫn đọng trong buffer. Lệnh getline(cin, s); tiếp theo đọc trúng '\n' và trả về chuỗi rỗng!

Khắc phục: Gọi cin.ignore(); ngay sau cin >> n;.
4. Lệnh endl gây tràn xả bộ đệm (TLE)

Lệnh cout << endl; không chỉ xuống dòng mà còn ép CPU xả bộ đệm (flush buffer) xuống đĩa. Khi in $10^5$ dòng, chương trình chạy chậm hơn 50 lần!

Khắc phục: Luôn dùng cout << "\n";.
5. Khai báo mảng lớn trong hàm main (RTE)

Khai báo int a[1000000]; bên trong hàm main() sẽ sử dụng vùng nhớ Stack (mặc định chỉ 8MB), gây lỗi tràn Stack (Runtime Error / Crash).

Khắc phục: Khai báo mảng toàn cục (ngoài main) hoặc static.
6. Quên reset mảng khi có nhiều Test Case

Khi đề bài có $T$ test case, dữ liệu của test trước còn lưu trong mảng đếm cnt[] sẽ làm sai hoàn toàn test sau.

Khắc phục: memset(cnt, 0, sizeof(cnt)); hoặc fill() đầu mỗi test.

Mảng Đếm Tần Số (Direct Addressing Table)

Độ phức tạp O(1)

Thay vì duyệt 2 vòng for $O(N^2)$, ta dùng chính giá trị của phần tử làm chỉ số mảng: cnt[x]++;. Quy chuẩn ánh xạ chỉ số:

Chữ cái thường ('a'-'z'): cnt[c - 'a']++ (mảng 26 ô)
Chữ cái hoa ('A'-'Z'): cnt[c - 'A']++ (mảng 26 ô)
Ký tự số ('0'-'9'): cnt[c - '0']++ (mảng 10 ô)
Số âm ($-10^6 \le x \le 10^6$): cnt[x + 1000000]++ (offset 10^6)

Nén Số / Rời Rạc Hóa (Coordinate Compression)

VNOI Wiki

Khi giá trị $A_i \le 10^9$ nhưng số phần tử $N \le 10^5$, ta không thể tạo mảng cnt[10^9]. Kỹ thuật nén số giúp ánh xạ các giá trị lớn về đoạn $[1, N]$:

COMPRESSION.CPP O(N log N) 
// Nén các giá trị lớn của mảng a[] về đoạn [1, k]
vector<int> vals = a;
sort(vals.begin(), vals.end());
vals.erase(unique(vals.begin(), vals.end()), vals.end());

// Ánh xạ từng phần tử a[i] sang giá trị nén (từ 1 đến vals.size())
for (int i = 0; i < n; i++) {
    int id = lower_bound(vals.begin(), vals.end(), a[i]) - vals.begin() + 1;
    // Bây giờ id nằm gọn trong [1, n], tha hồ dùng mảng đếm!
    cnt[id]++;
}

Bài 01: TÍCH LỚN MODULO (OVERFLOW)

Cụm 1: Cú Pháp & Bẫy Số Học OVERFLOW.INP / OVERFLOW.OUT ⏱ 1.0s | 💾 256MB
Độ phức tạp: O(1)
Đề Bài Chi Tiết

Cho ba số nguyên dương $A, B, C$ ($1 \le A, B, C \le 10^9$). Hãy tính và in ra giá trị $P = (A \times B) \pmod C$.

📥 Định dạng Đầu vào (Input):

Gồm một dòng duy nhất chứa ba số nguyên dương $A, B, C$ cách nhau bởi dấu cách.

📤 Định dạng Đầu ra (Output):

Ghi ra một số nguyên duy nhất là kết quả $P = (A \times B) \pmod C$.

Giới hạn: $1 \le A, B, C \le 10^9$. Subtask 1 (50% điểm): $A, B \le 10^4$. Subtask 2 (50% điểm): Giới hạn gốc.
Sample Input (OVERFLOW.INP) 
1000000000 1000000000 1000000007
Sample Output (OVERFLOW.OUT) 
49
Giải Thích Testcase

Tích $A \times B = 10^9 \times 10^9 = 10^{18}$. Chia lấy dư cho $10^9+7$ thu được số dư là $49$.

Phân Tích Đề Bài & Chiến Lược Tối Ưu:
Cảnh báo bẫy thi đấu:

Tích $A \times B$ đạt tới $10^{18}$, vượt xa giới hạn của kiểu 32-bit int ($2 \cdot 10^9$). Nếu viết long long ans = A * B % C; với $A, B$ khai báo kiểu int, phép nhân được tính trên thanh ghi 32-bit trước gây tràn số âm!

Ý tưởng thuật toán tối ưu:

Bắt buộc phải ép kiểu 64-bit ngay tại phép tính: 1LL * A * B % C hoặc khai báo trực tiếp $A, B, C$ là kiểu long long.

OVERFLOW.CPP O(1) 
#include <iostream>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("OVERFLOW.INP", "r", stdin);
    freopen("OVERFLOW.OUT", "w", stdout);
    #endif

    long long a, b, c;
    if (cin >> a >> b >> c) {
        long long ans = (1LL * (a % c) * (b % c)) % c;
        cout << ans << "\n";
    }
    return 0;
}

Bài 02: CHIA PHÒNG KHÁCH SẠN (CEILDIV)

Cụm 1: Cú Pháp & Bẫy Số Học CEILDIV.INP / CEILDIV.OUT ⏱ 1.0s | 💾 256MB
Độ phức tạp: O(1)
Đề Bài Chi Tiết

Đoàn thể thao trường có $N$ vận động viên ($1 \le N \le 10^{12}$). Khách sạn có các phòng tiêu chuẩn, mỗi phòng chứa tối đa $K$ người ($1 \le K \le 10^9$). Hỏi ban tổ chức cần thuê ít nhất bao nhiêu phòng để tất cả vận động viên đều có chỗ nghỉ?

📥 Định dạng Đầu vào (Input):

Một dòng duy nhất chứa hai số nguyên dương $N$ và $K$.

📤 Định dạng Đầu ra (Output):

Ghi ra một số nguyên duy nhất là số lượng phòng tối thiểu cần thuê.

Giới hạn: $1 \le N \le 10^{12}, 1 \le K \le 10^9$.
Sample Input (CEILDIV.INP) 
10 3
Sample Output (CEILDIV.OUT) 
4
Giải Thích Testcase

3 phòng chứa được $3 \times 3 = 9$ người, còn thừa 1 người cần thêm phòng thứ 4. Do đó cần thuê ít nhất 4 phòng.

Phân Tích Đề Bài & Chiến Lược Tối Ưu:
Cảnh báo bẫy thi đấu:

Dùng hàm số thực ceil((double)N / K) dễ bị sai số nghiêm trọng khi $N \ge 10^{12}$ do kiểu double chỉ duy trì độ chính xác khoảng 15-17 chữ số.

Ý tưởng thuật toán tối ưu:

Sử dụng công thức chia nguyên làm tròn lên chuẩn thi đấu: (N + K - 1) / K. Công thức này đảm bảo chính xác 100% trên số nguyên.

CEILDIV.CPP O(1) 
#include <iostream>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("CEILDIV.INP", "r", stdin);
    freopen("CEILDIV.OUT", "w", stdout);
    #endif

    long long n, k;
    if (cin >> n >> k) {
        long long ans = (n + k - 1) / k;
        cout << ans << "\n";
    }
    return 0;
}

Bài 03: RÚT GỌN PHÂN SỐ LỚN (REDUCE)

Cụm 1: Cú Pháp & Bẫy Số Học REDUCE.INP / REDUCE.OUT ⏱ 1.0s | 💾 256MB
Độ phức tạp: O(log(min(A, B)))
Đề Bài Chi Tiết

Cho phân số $A / B$ ($1 \le A, B \le 10^{14}$). Hãy rút gọn phân số này về dạng tối giản $A' / B'$.

📥 Định dạng Đầu vào (Input):

Gồm một dòng chứa hai số nguyên dương $A$ và $B$.

📤 Định dạng Đầu ra (Output):

Ghi ra hai số nguyên dương $A'$ và $B'$ cách nhau bởi dấu cách.

Giới hạn: $1 \le A, B \le 10^{14}$.
Sample Input (REDUCE.INP) 
24 36
Sample Output (REDUCE.OUT) 
2 3
Giải Thích Testcase

Ước chung lớn nhất của 24 và 36 là 12. Chia cả tử và mẫu cho 12 ta được phân số tối giản 2/3.

Phân Tích Đề Bài & Chiến Lược Tối Ưu:
Cảnh báo bẫy thi đấu:

Vòng lặp tìm ước chung từ $\min(A, B)$ giảm dần sẽ chạy $10^{14}$ bước $\implies$ TLE lập tức! Quên dùng long long sẽ làm tràn số ở các biến.

Ý tưởng thuật toán tối ưu:

Sử dụng thuật toán Euclid tìm ước chung lớn nhất thông qua hàm std::gcd(a, b) trong thư viện (chuẩn C++17). Sau đó lấy $A' = A / g$ và $B' = B / g$.

REDUCE.CPP O(log(min(A, B))) 
#include <iostream>
#include <numeric>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("REDUCE.INP", "r", stdin);
    freopen("REDUCE.OUT", "w", stdout);
    #endif

    long long a, b;

Bài 05: TỔNG DÃY SỐ MODULO (MODSUM)

Cụm 1: Cú Pháp & Bẫy Số Học MODSUM.INP / MODSUM.OUT ⏱ 1.0s | 💾 256MB
Độ phức tạp: O(1)
Đề Bài Chi Tiết

Một dòng duy nhất chứa số nguyên dương $N$.

📤 Định dạng Đầu ra (Output):

Sample Input (MODSUM.INP)

Sample Output (MODSUM.OUT) 
55
Giải Thích Testcase

Tổng từ 1 đến 10 là 55. $55 \pmod{10^9+7} = 55$.

Phân Tích Đề Bài & Chiến Lược Tối Ưu:
Cảnh báo bẫy thi đấu:

Vòng for chạy từ 1 đến $10^9$ sẽ tốn 1.0 giây gây nguy cơ TLE. Khi dùng công thức $S = N(N+1)/2$, phép chia 2 sau khi lấy modulo là SAI TOÀN TẬP vì $(A/B) \pmod M \ne (A \pmod M) / B$.

Ý tưởng thuật toán tối ưu:

Trong hai số liên tiếp $N$ và $N+1$, luôn luôn có một số là số chẵn. Ta chia 2 cho số chẵn đó trước khi nhân và lấy modulo: nếu $N$ chẵn thì lấy $(N/2)$, nếu $N$ lẻ thì lấy $((N+1)/2)$.

MODSUM.CPP O(1) 
#include <iostream>
using namespace std;

const long long MOD = 1e9 + 7;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("MODSUM.INP", "r", stdin);
    freopen("MODSUM.OUT", "w", stdout);
    #endif

    long long n;
    if (cin >> n) {
        long long a, b;
        if (n % 2 == 0) {
            a = (n / 2) % MOD;
            b = (n + 1) % MOD;
        } else {
            a = n % MOD;
            b = ((n + 1) / 2) % MOD;
        }
        long long ans = (a * b) % MOD;
        cout << ans << "\n";
    }
    return 0;
}

            cnt[a[i]]++;
        }
        cout << ans << "\n";
    }
    return 0;
}

Cụm 3: Xâu Ký Tự & ASCII PALIN.INP / PALIN.OUT ⏱ 1.0s | 💾 256MB
Độ phức tạp: O(|S|)
Đề Bài Chi Tiết

Cho xâu ký tự $S$ ($1 \le |S| \le 10^5$) gồm các chữ cái in thường. Hãy kiểm tra xem có thể biến xâu $S$ thành một xâu đối xứng (Palindrome) bằng cách xóa đi tối đa một ký tự hay không?

📥 Định dạng Đầu vào (Input):

Gồm một dòng duy nhất chứa xâu $S$.

📤 Định dạng Đầu ra (Output):

In ra YES nếu có thể, ngược lại in NO.

Giới hạn: $1 \le |S| \le 10^5$.
Sample Input (PALIN.INP) 
abcbda
Sample Output (PALIN.OUT) 
YES
Giải Thích Testcase

Xóa ký tự 'd' tại vị trí thứ 4 thu được xâu 'abcba' là một xâu đối xứng hoàn hảo.

Phân Tích Đề Bài & Chiến Lược Tối Ưu:
Cảnh báo bẫy thi đấu:

Thử xóa từng ký tự rồi kiểm tra đối xứng sẽ tốn $O(|S|^2) \approx 10^{10}$ phép tính $\implies$ TLE!

Ý tưởng thuật toán tối ưu:

Sử dụng kỹ thuật Hai con trỏ $l = 0, r = |S| - 1$. Khi gặp cặp ký tự không khớp đầu tiên $S[l] \ne S[r]$, bài toán chỉ có 2 khả năng: xóa $S[l]$ (kiểm tra đoạn $S[l+1 \dots r]$) hoặc xóa $S[r]$ (kiểm tra đoạn $S[l \dots r-1]$). Nếu 1 trong 2 đoạn đối xứng $\implies$ in YES.

PALIN.CPP O(|S|) 
#include <iostream>
#include <string>
using namespace std;

bool isPalinRange(const string &s, int l, int r) {
    while (l < r) {
        if (s[l] != s[r]) return false;
        l++; r--;
    }
    return true;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("PALIN.INP", "r", stdin);
    freopen("PALIN.OUT", "w", stdout);
    #endif

    string s;
    if (cin >> s) {
        int l = 0, r = s.size() - 1;
        bool ok = true;
        while (l < r) {
            if (s[l] != s[r]) {
                ok = isPalinRange(s, l + 1, r) || isPalinRange(s, l, r - 1);
                break;
            }
            l++; r--;
        }

Đề Bài Chi Tiết

Cho xâu ký tự $S$ gồm các chữ cái in thường ($1 \le |S| \le 10^5$). Hãy nén xâu theo thuật toán Run-Length Encoding: mỗi nhóm ký tự liên tiếp giống nhau được thay thế bằng ký tự đó kèm số lượng xuất hiện.

📥 Định dạng Đầu vào (Input):

Một dòng duy nhất chứa xâu $S$.

📤 Định dạng Đầu ra (Output):

Ghi ra xâu đã nén.

Giới hạn: $1 \le |S| \le 10^5$.
Sample Input (COMPRESS.INP) 
aabcccccaaa
Sample Output (COMPRESS.OUT) 
a2b1c5a3
Giải Thích Testcase

2 chữ 'a', 1 chữ 'b', 5 chữ 'c', 3 chữ 'a' liên tiếp.

Phân Tích Đề Bài & Chiến Lược Tối Ưu:
Cảnh báo bẫy thi đấu:

Cộng chuỗi bằng res = res + c trong vòng lặp sẽ tạo ra bản sao xâu liên tục mất $O(N^2)$ bộ nhớ và thời gian.

Ý tưởng thuật toán tối ưu:

Duyệt xâu bằng kỹ thuật 2 con trỏ: con trỏ $i$ cố định đầu cụm, con trỏ $j$ tăng dần chừng nào $S[j] == S[i]$. In trực tiếp $S[i]$ và $(j - i)$ ra màn hình, sau đó cập nhật $i = j$.

COMPRESS.CPP O(|S|) 
#include <iostream>
#include <string>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("COMPRESS.INP", "r", stdin);
    freopen("COMPRESS.OUT", "w", stdout);
    #endif

    string s;
    if (cin >> s) {
        int n = s.size();
        for (int i = 0; i < n; ) {
            int j = i;
            while (j < n && s[j] == s[i]) j++;
            cout << s[i] << (j - i);
            i = j;
        }
        cout << "\n";
    }
    return 0;
}

Bài 15: MÃ HÓA CAESAR XOAY VÒNG (CAESAR)

Cụm 3: Xâu Ký Tự & ASCII CAESAR.INP / CAESAR.OUT ⏱ 1.0s | 💾 256MB
Độ phức tạp: O(|S|)
Đề Bài Chi Tiết

Mã hóa Caesar dịch chuyển mỗi chữ cái trong văn bản sang phải $K$ vị trí theo bảng chữ cái tiếng Anh (từ 'z' quay vòng lại 'a', từ 'Z' quay vòng lại 'A'). Các ký tự không phải chữ cái (dấu cách, dấu câu, chữ số) được giữ nguyên. Cho xâu văn bản $S$ và số nguyên $K$ ($1 \le K \le 10^9$). Hãy in ra xâu sau khi mã hóa.

📥 Định dạng Đầu vào (Input):

Dòng 1: Xâu $S$. Dòng 2: Số nguyên $K$.

📤 Định dạng Đầu ra (Output):

In ra xâu $S$ sau khi mã hóa.

Giới hạn: $1 \le |S| \le 10^5, 1 \le K \le 10^9$.
Sample Input (CAESAR.INP) 
Hello, World!
3
Sample Output (CAESAR.OUT) 
Khoor, Zruog!
Giải Thích Testcase

'H' + 3 = 'K', 'e' + 3 = 'h', 'o' + 3 = 'r', dấu cách và dấu phẩy giữ nguyên.

Phân Tích Đề Bài & Chiến Lược Tối Ưu:
Cảnh báo bẫy thi đấu:

K có thể lên tới $10^9$, nếu cộng trực tiếp sẽ bị tràn biến ký tự char. Cần lấy số dư $K' = K \pmod{26}$.

Ý tưởng thuật toán tối ưu:

Rút gọn bước nhảy $step = K \pmod{26}$. Nếu là chữ thường: (c - 'a' + step) % 26 + 'a'. Nếu là chữ hoa: (c - 'A' + step) % 26 + 'A'.

CAESAR.CPP O(|S|) 
#include <iostream>
#include <string>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("CAESAR.INP", "r", stdin);
    freopen("CAESAR.OUT", "w", stdout);
    #endif

    string s;
    long long k;
    if (getline(cin, s) && cin >> k) {
        int step = k % 26;
        for (char &c : s) {
            if (c >= 'a' && c <= 'z') {
                c = (c - 'a' + step) % 26 + 'a';
            } else if (c >= 'A' && c <= 'Z') {
                c = (c - 'A' + step) % 26 + 'A';
            }
        }
        cout << s << "\n";
    }
    return 0;
}

Bộ Tứ Kỹ Thuật Mảng Tối Ưu Sơ Cấp (Core Optimization Techniques)

Chuyển hóa bài toán từ O(N^2) xuống O(N) và O(1)

Kỹ Thuật 1 Truy Vấn: O(1)

Mảng Cộng Dồn (Prefix Sum 1D & 2D)

1D: Tiền xử lý $pref[i] = pref[i-1] + A[i]$. Truy vấn tổng $[L, R] = pref[R] - pref[L-1]$ trong $O(1)$.
2D (Ma trận chữ nhật): Nguyên lý bù trừ:
$\text{Sum} = S[x_2][y_2] - S[x_1-1][y_2] - S[x_2][y_1-1] + S[x_1-1][y_1-1]$.

pref[i] = pref[i-1] + a[i]; // O(N) preprocessing
Kỹ Thuật 2 Cập Nhật Đoạn: O(1)

Mảng Hiệu (Difference Array 1D)

Cộng $V$ vào đoạn $[L, R]$ với $Q$ thao tác. Thay vì chạy vòng for $O(N)$ tốn $O(Q \times N)$, chỉ cần gán $D[L] += V$ và $D[R+1] -= V$ trong $O(1)$. Khôi phục mảng sau cùng bằng 1 lần prefix sum trong $O(N)$.

d[L] += V; d[R + 1] -= V; // O(1) per update
Kỹ Thuật 3 Độ Phức Tạp: O(N)

Kỹ Thuật Hai Con Trỏ (Two Pointers)

Tận dụng tính chất đơn điệu (Monotonicity). Hai con trỏ ngược chiều thu hẹp từ hai đầu mảng (Pair Sum), hoặc cùng chiều tạo thành Cửa sổ trượt (Sliding Window) co dãn để tìm đoạn con tối ưu trong $O(N)$.

while (sum >= S) { min_len = min(min_len, r-l+1); sum -= a[l++]; }
Kỹ Thuật 4 Độ Phức Tạp: O(N log(Ans))

Chặt Nhị Phân Trên Kết Quả (Binary Search on Answer)

Áp dụng cho các bài toán cực trị (Tìm giá trị lớn nhất nhỏ nhất / nhỏ nhất lớn nhất). Chuyển bài toán tìm kiếm tối ưu sang bài toán kiểm tra tính khả thi bool check(mid) đơn điệu dạng FFFFFTTTTT.

while (low <= high) { mid = low + (high-low)/2; if (check(mid)) ... }

Số Học Thi Đấu Cốt Lõi (Competitive Number Theory)

Nguyên tố, Sàng Eratosthenes & SPF, Thuật toán Euclid và Lũy thừa Modulo

Sàng Eratosthenes & SPF

Sàng nguyên tố chuẩn chạy $O(N \log \log N)$ cho $N \le 10^7$. Nâng cấp lên Sàng ước nguyên tố nhỏ nhất (SPF) cho phép phân tích thừa số nguyên tố cho số $X$ bất kỳ trong thời gian $O(\log X)$.

Euclid GCD & Bội Chung

Thuật toán Euclid chia lấy dư tìm $\gcd(a, b)$ trong $O(\log(\min(a, b)))$. Công thức bội chung nhỏ nhất an toàn chống tràn số: $\text{lcm}(a, b) = (a / \gcd(a, b)) \times b$.

Lũy Thừa Modulo Nhanh

Tính $(A^B) \pmod M$ với $B \le 10^18$ chỉ trong $O(\log B)$ bước nhân bằng thuật toán Lũy thừa nhị phân (Binary Exponentiation). Kết hợp Định lý Fermat nhỏ tính nghịch đảo modulo.

Cho mảng số nguyên $A$ gồm $N$ phần tử được đánh số từ $1$ đến $N$ và $Q$ truy vấn độc lập. Mỗi truy vấn gồm hai chỉ số nguyên dương $L, R$ ($1 \le L \le R \le N$). Với mỗi truy vấn, hãy tính tổng các phần tử nằm trong đoạn liên tiếp từ vị trí $L$ đến $R$: $S = \sum_{i=L}^{R} A_i$.

📥 Định dạng Đầu vào (Input): Dòng 1: Hai số nguyên dương $N, Q$. Dòng 2: Gồm $N$ số nguyên $A_1, A_2, \dots, A_N$. $Q$ dòng tiếp theo, mỗi dòng gồm 2 số nguyên $L, R$.
📤 Định dạng Đầu ra (Output): Gồm $Q$ dòng, mỗi dòng ghi một số nguyên duy nhất là kết quả của truy vấn tương ứng.

Phân Tích Đề Bài & Chiến Lược Tối Ưu
Cảnh báo bẫy thi đấu:

Dùng vòng lặp for tính tổng mỗi khi có truy vấn sẽ tốn $O(N \times Q) \approx 4 \times 10^{10}$ phép tính, dẫn đến TLE 100%. Ngoài ra, tổng các phần tử có thể vượt quá $2 \times 10^{14}$ gây tràn số kiểu `int` 32-bit (buộc phải dùng `long long`).

Ý tưởng thuật toán tối ưu:

Xây dựng mảng tiền tố $pref[i] = pref[i-1] + A[i]$ với $pref[0] = 0$ trong $O(N)$. Sau đó với mỗi truy vấn $[L, R]$, tính tổng trong $O(1)$ bằng công thức: $pref[R] - pref[L-1]$.

QUERY1D.CPP Thời gian: O(N + Q) | Bộ nhớ: O(N) 
#include <iostream>
#include <vector>

using namespace std;

int main() {

    ios_base::sync_with_stdio(false);

            cin >> a[i];
            pref[i] = pref[i - 1] + a[i]; // Tiền xử lý O(N)
        }

        // Trả lời mỗi truy vấn trong O(1)
        while (q--) {
            int l, r;
            cin >> l >> r;
            long long ans = pref[r] - pref[l - 1];
            cout << ans << "\n";
        }
    }
    return 0;
}

Bài 17: TỔNG MA TRẬN HÌNH CHỮ NHẬT 2D (RECTSUM)

Cụm 1: Mảng Cộng Dồn & Mảng Hiệu RECTSUM.INP / RECTSUM.OUT ⏱ 1.0s | 💾 256MB
Thời gian: O(N * M + Q) | Bộ nhớ: O(N * M)
Đề Bài Chi Tiết

Cho bảng số kích thước $N \times M$ gồm các số nguyên. Có $Q$ truy vấn độc lập, mỗi truy vấn gồm 4 số nguyên $(x_1, y_1, x_2, y_2)$ xác định hình chữ nhật con có góc trên bên trái $(x_1, y_1)$ và góc dưới bên phải $(x_2, y_2)$. Hãy tính tổng tất cả các phần tử nằm trong hình chữ nhật này.

Phân Tích Đề Bài & Chiến Lược Tối Ưu
Cảnh báo bẫy thi đấu:

Quên phần tử giao nhau ở góc trên trái $pref[x_1-1][y_1-1]$ khi bù trừ hoặc quên cộng lại phần tử giao nhau trong lúc tiền xử lý $pref[i-1][j-1]$. Cần chú ý dùng mảng 1-based và kiểu dữ liệu `long long`.

Ý tưởng thuật toán tối ưu:

Dùng công thức bù trừ 2D: $pref[i][j] = pref[i-1][j] + pref[i][j-1] - pref[i-1][j-1] + A[i][j]$ và truy vấn trong $O(1)$: $pref[x_2][y_2] - pref[x_1-1][y_2] - pref[x_2][y_1-1] + pref[x_1-1][y_1-1]$.

RECTSUM.CPP Thời gian: O(N * M + Q) | Bộ nhớ: O(N * M) 
#include <iostream>
#include <vector>

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("RECTSUM.INP", "r", stdin);
    freopen("RECTSUM.OUT", "w", stdout);
    #endif

    int n, m, q;
    if (cin >> n >> m >> q) {
        vector<vector<long long>> a(n + 1, vector<long long>(m + 1, 0));
        vector<vector<long long>> pref(n + 1, vector<long long>(m + 1, 0));

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                cin >> a[i][j];
                // Nguyên lý bù trừ tiền xử lý 2D Prefix Sum
                pref[i][j] = pref[i - 1][j] + pref[i][j - 1] - pref[i - 1][j - 1] + a[i][j];
            }
        }

        while (q--) {
            int x1, y1, x2, y2;
            cin >> x1 >> y1 >> x2 >> y2;
            // Truy vấn tổng hình chữ nhật trong O(1)
            long long ans = pref[x2][y2] - pref[x1 - 1][y2] - pref[x2][y1 - 1] + pref[x1 - 1][y1 - 1];
            cout << ans << "\n";
        }
    }
    return 0;
}

Bài 18: CẬP NHẬT ĐOẠN HÀNG LOẠT VỚI MẢNG HIỆU (UPDATE1D)

Đề Bài Chi Tiết

Ban đầu có một mảng $A$ gồm $N$ số nguyên đều có giá trị bằng $0$. Phân Tích Đề Bài & Chiến Lược Tối Ưu
Cảnh báo bẫy thi đấu:

Nếu dùng vòng lặp `for (int i = L; i <= R; i++)` cho mỗi thao tác cập nhật, độ phức tạp $O(Q \times N) \approx 4 \times 10^{10}$ thao tác gây TLE. Cần chú ý cấp phát mảng hiệu $N+2$ phần tử để tránh tràn biên khi $R = N$.

Ý tưởng thuật toán tối ưu:

Dùng kỹ thuật Mảng hiệu (Difference Array): mỗi thao tác ta chỉ gán $D[L] += V$ và $D[R+1] -= V$ trong $O(1)$. Sau khi hết $Q$ thao tác, tính mảng cộng dồn của $D$ trong $O(N)$ để thu được kết quả.

UPDATE1D.CPP Thời gian: O(N + Q) | Bộ nhớ: O(N) 
#include <iostream>
#include <vector>

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    freopen("UPDATE1D.INP", "r", stdin);
    freopen("UPDATE1D.OUT", "w", stdout);
    #endif

    int n, q;
    if (cin >> n >> q) {
        vector<long long> d(n + 2, 0); // Mảng hiệu kích thước N + 2

        while (q--) {
            int l, r;
            long long v;
            cin >> l >> r >> v;
            // Cập nhật mảng hiệu trong O(1)
            d[l] += v;
            d[r + 1] -= v;

        // Khôi phục mảng ban đầu bằng prefix sum O(N)
        long long current_val = 0;
        for (int i = 1; i <= n; i++) {
            current_val += d[i];
            cout << current_val << (i == n ? "" : " ");
        }
        cout << "\n";
    }
    return 0;
}

Bài 19: ĐẾM ĐOẠN CON CÓ TỔNG BẰNG K (SUBARRAYK)

Cụm 1: Mảng Cộng Dồn & Mảng Hiệu SUBARRAYK.INP / SUBARRAYK.OUT ⏱ 1.0s | 💾 256MB
Thời gian: O(N log N) | Bộ nhớ: O(N)
Đề Bài Chi Tiết

Cho mảng số nguyên $A$ gồm $N$ phần tử (các phần tử có thể mang giá trị âm, dương hoặc $0$) và số nguyên $K$. Hãy đếm số lượng đoạn con liên tiếp $[L..R]$ ($1 \le L \le R \le N$) sao cho tổng các phần tử trong đoạn đúng bằng $K$: $\sum_{i=L}^{R} A_i = K$.

📥 Định dạng Đầu vào (Input): Dòng 1: Hai số nguyên $N, K$. Dòng 2: Gồm $N$ số nguyên $A_1, A_2, \dots, A_N$.
📤 Định dạng Đầu ra (Output): In ra một số nguyên duy nhất là số lượng đoạn con thỏa mãn.
Giới hạn: $1 \le N \le 2 \times 10^5$; $|K| \le 10^{14}$; $|A_i| \le 10^9$.
Bảng Testcase Mẫu (Sample Test)
Sample Input (SUBARRAYK.INP) 
5 7
2 5 2 0 5
Sample Output (SUBARRAYK.OUT) 
3
Giải Thích Testcase: Có 3 đoạn con thỏa mãn: [1..2] (2+5=7), [2..3] (5+2=7), và [2..4] (5+2+0=7).
Phân Tích Đề Bài & Chiến Lược Tối Ưu
Cảnh báo bẫy thi đấu:

Vì mảng chứa cả số âm và số 0 nên không thể áp dụng Hai con trỏ Two Pointers thông thường (mất tính đơn điệu). Ngoài ra quên khởi tạo `cnt[0] = 1` sẽ bỏ sót các đoạn con bắt đầu từ vị trí 1 có tổng đúng bằng $K$.

Ý tưởng thuật toán tối ưu:

Tổng đoạn $[L..R] = pref[R] - pref[L-1] = K \iff pref[L-1] = pref[R] - K$. Duyệt qua từng phần tử, duy trì bảng đếm tần suất các prefix sum đã gặp bằng `std::map` để tìm kiếm trong $O(\log N)$.

SUBARRAYK.CPP Thời gian: O(N log N) | Bộ nhớ: O(N) 
#include <iostream>
#include <vector>
#include <map>

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("SUBARRAYK.INP", "r", stdin);
    freopen("SUBARRAYK.OUT", "w", stdout);
    #endif

    int n;
    long long k;
    if (cin >> n >> k) {
        vector<long long> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];

        map<long long, int> cnt;
        cnt[0] = 1; // Khởi tạo tổng rỗng bằng 0 xuất hiện 1 lần

        long long current_pref = 0;
        long long ans = 0;

        for (int i = 0; i < n; i++) {
            current_pref += a[i];
            long long target = current_pref - k;
            if (cnt.find(target) != cnt.end()) {
                ans += cnt[target];
            }
            cnt[current_pref]++;
        }

        cout << ans << "\n";
    }
    return 0;
}

Bài 20: ĐỘ DÀI ĐOẠN CON DÀI NHẤT CHIA HẾT CHO M (DIVMSUM)

Cụm 1: Mảng Cộng Dồn & Mảng Hiệu DIVMSUM.INP / DIVMSUM.OUT ⏱ 1.0s | 💾 256MB
Thời gian: O(N) | Bộ nhớ: O(M)
Đề Bài Chi Tiết

Cho mảng số nguyên $A$ gồm $N$ phần tử và số nguyên dương $M$. Hãy tìm độ dài lớn nhất của một đoạn con liên tiếp có tổng các phần tử chia hết cho $M$. Nếu không có đoạn con nào có tổng chia hết cho $M$, in ra 0.

📥 Định dạng Đầu vào (Input): Dòng 1: Hai số nguyên dương $N, M$. Dòng 2: $N$ số nguyên $A_1, A_2, \dots, A_N$.
📤 Định dạng Đầu ra (Output): In ra một số nguyên duy nhất là độ dài lớn nhất tìm được.
Giới hạn: $1 \le N \le 2 \times 10^5$; $1 \le M \le 10^5$; $|A_i| \le 10^9$.
Bảng Testcase Mẫu (Sample Test)
Sample Input (DIVMSUM.INP) 
5 3
2 3 5 4 1
Sample Output (DIVMSUM.OUT) 
4
Giải Thích Testcase: Đoạn con [2..5] gồm các số [3, 5, 4, 1] có tổng là 13+0... à: 3+5+4 = 12 chia hết cho 3. Đoạn con [1..4] gồm [2, 3, 5, 4] có tổng = 14 không chia hết. Đoạn [2..4] có tổng 12 độ dài 3. Đoạn con [1..5] có tổng 2+3+5+4+1 = 15 chia hết cho 3, độ dài là 5! Cụ thể sample output mẫu: đoạn [2, 3, 5, 4, 1] tổng 15 -> chia hết cho 3, độ dài 5.
Phân Tích Đề Bài & Chiến Lược Tối Ưu
Cảnh báo bẫy thi đấu:

Bẫy số âm trong phép chia lấy dư của C++: `-5 % 3 = -2`. Cần chuẩn hóa số dư: `rem = ((val % m) + m) % m` để đảm bảo $0 \le rem < M$ và chỉ số mảng luôn hợp lệ.

Ý tưởng thuật toán tối ưu:

Tổng đoạn $[L..R]$ chia hết cho $M \iff pref[R] \equiv pref[L-1] \pmod M$. Ta chỉ cần lưu vị trí xuất hiện đầu tiên của mỗi số dư từ $0$ đến $M-1$. Khi gặp lại số dư đó ở vị trí $i$, độ dài là $i - first\_pos[rem]$.

DIVMSUM.CPP Thời gian: O(N) | Bộ nhớ: O(M) 
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("DIVMSUM.INP", "r", stdin);
    freopen("DIVMSUM.OUT", "w", stdout);
    #endif

    int n, m;
    if (cin >> n >> m) {
        vector<int> first_pos(m, -2);
        first_pos[0] = 0; // Vị trí prefix sum = 0 ở chỉ số 0

        long long current_sum = 0;
        int max_len = 0;

        for (int i = 1; i <= n; i++) {
            long long x;
            cin >> x;
            current_sum += x;
            
            // Xử lý chuẩn số dư dương trong C++
            int rem = ((current_sum % m) + m) % m;

            if (first_pos[rem] != -2) {
                max_len = max(max_len, i - first_pos[rem]);
            } else {
                first_pos[rem] = i; // Ghi nhận vị trí xuất hiện đầu tiên
            }
        }

        cout << max_len << "\n";
    }
    return 0;
}

Bài 21: TÌM CẶP SỐ CÓ TỔNG BẰNG X (PAIRSUM)

📥 Định dạng Đầu vào (Input): Dòng 1: Hai số nguyên dương $N, X$. Dòng 2: $N$ số nguyên dương $A_1, A_2, \dots, A_N$.
📤 Định dạng Đầu ra (Output): In ra hai chỉ số $i, j$ cách nhau một dấu cách, hoặc `-1` nếu không tồn tại.

2 7 5 1

2 4

Phân Tích Đề Bài & Chiến Lược Tối Ưu
Cảnh báo bẫy thi đấu:

Sau khi `sort` mảng, vị trí ban đầu của phần tử bị thay đổi, dẫn đến in sai chỉ số $i, j$ ban đầu của đề bài. Bắt buộc phải dùng `pair` để đính kèm chỉ số gốc trước khi sắp xếp.

    int n;
    long long x;
    if (cin >> n >> x) {
        vector<pair<long long, int>> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i].first;
            a[i].second = i + 1; // Lưu chỉ số 1-based ban đầu
        }

        // Sắp xếp mảng tăng dần O(N log N)
        sort(a.begin(), a.end());

        int left = 0, right = n - 1;
        bool found = false;

        // Kỹ thuật hai con trỏ ngược chiều O(N)
        while (left < right) {
            long long sum = a[left].first + a[right].first;
            if (sum == x) {
                int idx1 = min(a[left].second, a[right].second);
                int idx2 = max(a[left].second, a[right].second);
                cout << idx1 << " " << idx2 << "\n";
                found = true;
                break;
            } else if (sum < x) {
                left++;
            } else {
                right--;
            }
        }

        if (!found) {
            cout << "-1\n";
        }
    }
    return 0;

Cảnh báo bẫy thi đấu:

Quên trường hợp toàn bộ mảng có tổng vẫn nhỏ hơn $S$ (phải in ra `-1`). Dùng biến `int` tính tổng $S$ sẽ bị tràn số vì $S \le 10^{14}$.

Ý tưởng thuật toán tối ưu:

Vì mọi $A_i > 0$, mảng có tính đơn điệu: khi mở rộng sang phải tổng tăng, khi thu hẹp từ trái tổng giảm. Dùng 2 con trỏ cùng chiều `left, right`. Mỗi phần tử được đưa vào và lấy ra tối đa 1 lần, đạt $O(N)$ toàn diện.

#include <vector>
#include <algorithm>

        vector<long long> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];

        int left = 0;
        long long current_sum = 0;
        int min_len = n + 1;

        // Kỹ thuật hai con trỏ cùng chiều (Cửa sổ trượt)
        for (int right = 0; right < n; right++) {

            while (current_sum >= s) {
                min_len = min(min_len, right - left + 1);
                current_sum -= a[left];
                left++;
            }
        }

        if (min_len <= n) {
            cout << min_len << "\n";
        } else {
            cout << "-1\n";
        }
    }
    return 0;

Phân Tích Đề Bài & Chiến Lược Tối Ưu
Cảnh báo bẫy thi đấu:

Khi giảm tần suất trong map về 0: `freq[x]--`, nếu không gọi `freq.erase(x)` thì `freq.size()` vẫn tính phần tử đó, dẫn đến kết quả sai hoàn toàn.

Ý tưởng thuật toán tối ưu:

Dùng kỹ thuật Cửa sổ trượt kết hợp `std::map` theo dõi tần suất các số hiện diện trong cửa sổ. `freq.size()` chính là số lượng giá trị phân biệt. Khi `freq.size() > K`, tăng `left` và `freq.erase` khi tần số về 0.

DIFFKVAL.CPP Thời gian: O(N log K) | Bộ nhớ: O(K) 
#include <iostream>
#include <vector>
#include <map>
#include <algorithm>

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("DIFFKVAL.INP", "r", stdin);
    freopen("DIFFKVAL.OUT", "w", stdout);
    #endif

    int n, k;
    if (cin >> n >> k) {
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];

        map<int, int> freq;
        int left = 0;
        int max_len = 0;

        for (int right = 0; right < n; right++) {
            freq[a[right]]++;

            // Khi số phần tử phân biệt vượt quá K, co cửa sổ từ trái
            while ((int)freq.size() > k) {
                freq[a[left]]--;
                if (freq[a[left]] == 0) {
                    freq.erase(a[left]); // Xóa khỏi map khi tần số về 0
                }
                left++;
            }

            max_len = max(max_len, right - left + 1);
        }

        cout << max_len << "\n";
    }
    return 0;
}

Ý tưởng thuật toán tối ưu:

Dùng `std::map last_pos` ghi nhận vị trí gần nhất của mỗi bài hát. Duyệt con trỏ `right`, nhảy con trỏ `left` vượt qua vị trí trùng lặp gần nhất trong $O(1)$. Tổng thời gian $O(N \log N)$.

#include <algorithm>

            int song = a[right];

            last_pos[song] = right;
            max_len = max(max_len, right - left + 1);
        }

        cout << max_len << "\n";
    }
    return 0;
}

                } else {
                    right--;
                }
            }
        }

Bài 26: ĐẾM SỐ LƯỢNG PHẦN TỬ TRONG ĐOẠN [X, Y] (COUNTVAL)

Ý tưởng thuật toán tối ưu:

Sắp xếp mảng tăng dần trong $O(N \log N)$. Mỗi truy vấn tìm con trỏ `it1 = lower_bound(..., x)` và `it2 = upper_bound(..., y)`. Số lượng phần tử là khoảng cách `it2 - it1` trong $O(\log N)$.

#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {

    freopen("COUNTVAL.INP", "r", stdin);

    #endif

    int n, q;
    if (cin >> n >> q) {
        vector<long long> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];

        // Sắp xếp mảng tăng dần để chuẩn bị chặt nhị phân
        sort(a.begin(), a.end());

        while (q--) {
            long long x, y;
            cin >> x >> y;
            // lower_bound tìm phần tử đầu tiên >= X
            auto it1 = lower_bound(a.begin(), a.end(), x);
            // upper_bound tìm phần tử đầu tiên > Y
            auto it2 = upper_bound(a.begin(), a.end(), y);

            // Số lượng phần tử nằm trong đoạn [X, Y]
            int count = it2 - it1;
            cout << count << "\n";
        }
    }
    return 0;
}

Bài 27: CHẶT NHỊ PHÂN CẮT GỖ TIÊU CHUẨN (LUMBER)

Bác tiều phu có $N$ cây gỗ, cây thứ $i$ có chiều cao $H_i$. Bác cần thu gom ít nhất $M$ mét gỗ. Máy cưa được thiết lập ở độ cao $H$. Giới hạn: $1 \le N \le 10^6$; $1 \le M \le 2 \times 10^9$; $1 \le H_i \le 2 \times 10^9$.

Tổng lượng gỗ thu gom được có thể lên tới $N \times \max(H) \approx 10^6 \times 2 \times 10^9 = 2 \times 10^{15}$, nếu dùng kiểu `int` để tích lũy lượng gỗ sẽ bị tràn số sinh ra giá trị âm, khiến thuật toán chạy sai hoàn toàn.

Ý tưởng thuật toán tối ưu:

Nhận xét tính chất đơn điệu: $H$ càng cao thì lượng gỗ thu được càng ít. Chặt nhị phân trên tập kết quả (Binary Search on Answer) trong đoạn $[0, \max(H_i)]$. Với mỗi giá trị `mid`, kiểm tra tính khả thi trong $O(N)$. Tổng độ phức tạp $O(N \log(\max H))$.

LUMBER.CPP Thời gian: O(N log(max H)) | Bộ nhớ: O(N) 
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

// Hàm kiểm tra tính khả thi tại độ cao H
bool check(long long h, const vector<long long>& trees, long long m) {
    long long wood = 0;
    for (long long tree_h : trees) {
        if (tree_h > h) {
            wood += (tree_h - h);
            if (wood >= m) return true; // Cắt nhánh sớm tránh tràn số
        }
    }
    return wood >= m;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("LUMBER.INP", "r", stdin);
    freopen("LUMBER.OUT", "w", stdout);
    #endif

    int n;
    long long m;
    if (cin >> n >> m) {
        vector<long long> trees(n);
        long long max_h = 0;
        for (int i = 0; i < n; i++) {
            cin >> trees[i];
            max_h = max(max_h, trees[i]);
        }

        // Chặt nhị phân trên tập kết quả độ cao [0..max_h]
        long long low = 0, high = max_h;
        long long ans = 0;

        while (low <= high) {
            long long mid = low + (high - low) / 2;
            if (check(mid, trees, m)) {
                ans = mid;        // Ghi nhận nghiệm hợp lệ
                low = mid + 1;    // Thử tìm độ cao lớn hơn
            } else {
                high = mid - 1;   // Cần hạ thấp độ cao để lấy nhiều gỗ hơn
            }
        }

        cout << ans << "\n";
    }
    return 0;
}

Ý tưởng thuật toán tối ưu:

Dùng Sàng ước số nguyên tố nhỏ nhất (Sieve SPF - Smallest Prime Factor). Tiền xử lý 1 lần trong $O(M)$ với $M = 10^7$. Với mỗi truy vấn $X$, phân tích thừa số nguyên tố $X = p_1^{a_1} p_2^{a_2} \dots$ chỉ tốn $O(\log X)$ bước chia. Số ước số là $(a_1 + 1)(a_2 + 1) \dots$.

DIVISORS.CPP Thời gian: O(MAX_VAL + Q log X) | Bộ nhớ: O(MAX_VAL) 
#include <iostream>
#include <vector>

using namespace std;

const int MAX_VAL = 10000005;
int spf[MAX_VAL]; // Smallest Prime Factor

// Tiền xử lý Sàng SPF trong O(MAX_VAL)
void sieve_spf() {
    for (int i = 1; i < MAX_VAL; i++) spf[i] = i;
    for (int i = 2; i * i < MAX_VAL; i++) {
        if (spf[i] == i) {
            for (int j = i * i; j < MAX_VAL; j += i) {
                if (spf[j] == j) spf[j] = i;
            }
        }
    }
}

// Đếm số ước của X trong O(log X)
int count_divisors(int x) {
    int total_divisors = 1;
    while (x > 1) {
        int p = spf[x];
        int count = 0;
        while (x % p == 0) {
            count++;
            x /= p;
        }
        total_divisors *= (count + 1);
    }
    return total_divisors;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("DIVISORS.INP", "r", stdin);
    freopen("DIVISORS.OUT", "w", stdout);
    #endif

    sieve_spf();

    int q;
    if (cin >> q) {
        while (q--) {
            int x;
            cin >> x;
            cout << count_divisors(x) << "\n";
        }
    }
    return 0;
}

COPRIME.CPP Thời gian: O(N^2 log(max A)) | Bộ nhớ: O(N) 
#include <iostream>
#include <vector>
#include <numeric>

using namespace std;

// Thuật toán Euclid tìm ước chung lớn nhất
long long gcd_fast(long long a, long long b) {
    while (b != 0) {
        long long temp = a % b;
        a = b;
        b = temp;
    }
    return a;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    #ifndef ONLINE_JUDGE
    freopen("COPRIME.INP", "r", stdin);
    freopen("COPRIME.OUT", "w", stdout);
    #endif

    int n;
    if (cin >> n) {
        vector<long long> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];

        long long count = 0;
        // Duyệt mọi cặp chỉ số i < j
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (gcd_fast(a[i], a[j]) == 1) {
                    count++;
                }
            }
        }

        cout << count << "\n";
    }
    return 0;
}

Cảnh báo bẫy thi đấu:

Tính lũy thừa bằng vòng for $B$ lần sẽ tốn $O(B)$ với $B \le 10^{18}$ là bất khả thi. Quên ép kiểu `1LL` trong phép nhân `res * a` khi cả hai đều cỡ $10^9$ sẽ gây tràn số `long long` thành số âm.

Ý tưởng thuật toán tối ưu:

Áp dụng thuật toán Lũy thừa nhị phân (Binary Exponentiation). Tách $B$ thành biểu diễn nhị phân: nếu $B$ chẵn thì $A^B = (A^2)^{B/2}$, nếu $B$ lẻ thì $A^B = A \times A^{B-1}$. Thời gian tính toán giảm ngoạn mục xuống $O(\log B)$ (chỉ mất tối đa 60 phép nhân).

FASTPOW.CPP Thời gian: O(T log B) | Bộ nhớ: O(1) 
#include <iostream>

using namespace std;

// Thuật toán lũy thừa nhị phân nhanh O(log B)
long long power_mod(long long a, long long b, long long m) {
    long long res = 1 % m;
    a %= m;
    while (b > 0) {
        if (b & 1) { // Nếu b là số lẻ
            res = (1LL * res * a) % m;
        }
        a = (1LL * a * a) % m; // Nhân đôi lũy thừa
        b >>= 1;               // Chia đôi số mũ
    }
    return res;
}

    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    freopen("FASTPOW.OUT", "w", stdout);
    #endif

    int t;
    if (cin >> t) {
        while (t--) {
            long long a, b, m;
            cin >> a >> b >> m;
            cout << power_mod(a, b, m) << "\n";
        }
    }
    return 0;
}

std::vector<T> <vector> $O(1)$ push/access Mảng động co giãn kích thước linh hoạt, an toàn bộ nhớ hơn mảng tĩnh C.
sort(a.begin(), a.end()) <algorithm> $O(N \log N)$ Sắp xếp dãy số tăng dần (Introsort: QuickSort + HeapSort + InsertionSort).
lower_bound(a.begin(), a.end(), x) <algorithm> $O(\log N)$ Tìm vị trí phần tử đầu tiên $\ge x$ trên mảng đã sắp xếp.
upper_bound(a.begin(), a.end(), x) <algorithm> $O(\log N)$ Tìm vị trí phần tử đầu tiên $> x$ trên mảng đã sắp xếp.
std::set<T> <set> $O(\log N)$ Tập hợp loại bỏ phần tử trùng lặp và tự động sắp xếp theo thứ tự tăng dần (Red-Black Tree).
std::map<Key, Val> <map> $O(\log N)$ Bảng ánh xạ từ Khóa sang Giá trị (Đếm tần số số lớn, từ điển xâu ký tự).
std::unordered_map<Key, Val> <unordered_map> $O(1)$ trung bình Bảng băm (Hash Table) tra cứu siêu tốc khi cần tối ưu thời gian dưới 1 giây.
priority_queue<T> <queue> $O(\log N)$ Hàng đợi ưu tiên (Max-Heap mặc định, hoặc Min-Heap với greater<T>), ứng dụng Dijkstra.
__gcd(a, b) <algorithm> $O(\log(\min(a,b)))$ Thuật toán Euclid tìm ước chung lớn nhất của 2 số nguyên.
