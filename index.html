Bản Chất Mảng Đếm Tần Số (Direct Addressing Table)

Thay vì dùng 2 vòng for lồng nhau $O(N^2)$ để duyệt và đếm số lần xuất hiện của từng phần tử, ta sử dụng chính giá trị của phần tử làm chỉ số mảng:
cnt[x]++; $\implies$ Thao tác kiểm tra hoặc cập nhật đạt độ phức tạp $O(1)$!

Quy chuẩn ánh xạ mảng đếm theo kiểu dữ liệu:
  • Chữ cái thường ('a'-'z'): cnt[s[i] - 'a']++ (mảng 26 ô)
  • Chữ cái hoa ('A'-'Z'): cnt[s[i] - 'A']++ (mảng 26 ô)
  • Ký tự chữ số ('0'-'9'): cnt[s[i] - '0']++ (mảng 10 ô)
  • Mọi ký tự mã ASCII: cnt[(unsigned char)s[i]]++ (256 ô)

Ví Dụ Minh Họa Từng Bước: Bài Toán PAIRSUM ($A_i + A_j = K$)

Duyệt mảng một lượt $O(N)$ kết hợp mảng đếm tần số để đếm số cặp có tổng bằng $K = 8$ trong mảng $A = [3, 5, 2, 5, 4]$

Bước ($i$) Giá trị $A[i]$ Target cần tìm ($8 - A[i]$) Đã gặp trước đó (cnt[target]) Cộng dồn đáp án (ans += cnt) Cập nhật mảng đếm (cnt[A[i]]++) Ý nghĩa sư phạm
1 3 8 - 3 = 5 cnt[5] = 0 ans = 0 cnt[3] = 1 Chưa có số 5 nào xuất hiện trước số 3.
2 5 8 - 5 = 3 cnt[3] = 1 ans = 0 + 1 = 1 cnt[5] = 1 Ghép được cặp (3, 5) thứ nhất!
3 2 8 - 2 = 6 cnt[6] = 0 ans = 1 cnt[2] = 1 Chưa có số 6 nào xuất hiện trước số 2.
4 5 8 - 5 = 3 cnt[3] = 1 ans = 1 + 1 = 2 cnt[5] = 2 Ghép được cặp (3, 5) thứ hai!
5 4 8 - 4 = 4 cnt[4] = 0 ans = 2 cnt[4] = 1 Số 4 chưa từng xuất hiện trước đó.
Kết luận: Tìm ra chính xác 2 cặp chỉ bằng duy nhất 1 lần duyệt mảng. Độ phức tạp: $O(N)$ thời gian, $O(M)$ không gian bộ nhớ. Tối ưu 100% test

Code Mẫu 1: Đếm Tần Số & Ký Tự Duy Nhất 

// Đếm tần số ký tự thường và tìm ký tự xuất hiện 1 lần đầu tiên
#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    string s;
    if (!(cin >> s)) return 0;

    vector<int> cnt(26, 0);
    // Bước 1: Đếm tần số trong O(N)
    for (char c : s) {
        if (c >= 'a' && c <= 'z') cnt[c - 'a']++;
    }

    // Bước 2: Tìm ký tự duy nhất xuất hiện đầu tiên
    char firstUnique = '';
    for (char c : s) {
        if (c >= 'a' && c <= 'z' && cnt[c - 'a'] == 1) {
            firstUnique = c;
            break;
        }
    }

    if (firstUnique != '') 
        cout << "Ky tu duy nhat dau tien: " << firstUnique << "
";
    else 
        cout << "Khong co ky tu duy nhat
";
    return 0;
}

Code Mẫu 2: Giải PAIRSUM Trong O(N) 

// Đếm số cặp A[i] + A[j] = K trong O(N) bằng unordered_map (hoặc mảng tĩnh)
#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n; long long k;
    if (!(cin >> n >> k)) return 0;

    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    unordered_map<long long, int> cnt;
    long long totalPairs = 0;

    for (int i = 0; i < n; i++) {
        long long target = k - a[i];
        if (cnt.find(target) != cnt.end()) {
            totalPairs += cnt[target]; // Cộng dồn số lượng số target đã thấy
        }
        cnt[a[i]]++; // Cập nhật số hiện tại vào mảng đếm
    }

    cout << totalPairs << "
";
    return 0;
}

Bản Chất Mảng Cộng Dồn 1D (Prefix Sum)

Mảng cộng dồn $P$ lưu tổng của tất cả các phần tử từ chỉ số 1 đến chỉ số $i$.
Ta khởi tạo $P[0] = 0$ và xây dựng mảng trong $O(N)$ bằng công thức truy hồi:
P[i] = P[i - 1] + A[i];

Công thức trả lời mọi truy vấn tính tổng đoạn [L, R] trong O(1):
Sum(L, R) = P[R] - P[L - 1]

Quy tắc thi đấu sống còn: Luôn dùng 1-based index (chỉ số mảng bắt đầu từ 1). Khi $L = 1$, ta tính $P[1 - 1] = P[0] = 0$ một cách tự nhiên mà không bao giờ bị dính lỗi Runtime Error truy xuất $P[-1]$.

1.2

Mở Rộng: Prefix Sum 2D (Ma Trận Hình Chữ Nhật)

Cho bảng số $N imes M$. Ta muốn tính tổng mọi phần tử trong hình chữ nhật con từ góc trái trên $(x_1, y_1)$ đến góc phải dưới $(x_2, y_2)$ trong đúng $O(1)$ phép tính.

1. Công thức xây dựng bảng P[i][j]: P[i][j] = P[i-1][j] + P[i][j-1] - P[i-1][j-1] + A[i][j];
2. Công thức truy vấn O(1) hình chữ nhật: Sum = P[x2][y2] - P[x1-1][y2] - P[x2][y1-1] + P[x1-1][y1-1];
1.3

Ví Dụ Minh Họa Từng Bước: Xây Dựng Mảng Cộng Dồn & Truy Vấn Đoạn

Cho mảng $A = [3, 5, 2, 8, 4, 7, 1, 6]$ gồm $N = 8$ phần tử (1-based index). Tính trước mảng $P$ và truy vấn tổng đoạn $[3, 6]$

Chỉ số ($i$) 0 1 2 3 (L) 4 5 6 (R) 7 8
Mảng gốc $A[i]$ - 3 5 2 8 4 7 1 6
Prefix Sum $P[i]$ 0 3 8 (P[L-1]) 10 18 22 29 (P[R]) 30 36
Thao tác tính tổng đoạn [L = 3, R = 6]:
Cách tính trâu (Vòng lặp For): Sum = A[3] + A[4] + A[5] + A[6] = 2 + 8 + 4 + 7 = 21 Tốn $R - L + 1 = 4$ phép cộng $\implies O(N)$ mỗi truy vấn.
Cách tính bằng Prefix Sum: Sum = P[6] - P[3 - 1] = P[6] - P[2] = 29 - 8 = 21 Đúng duy nhất 1 phép trừ $\implies O(1)$ tức thì!

Code Mẫu 1: Prefix Sum 1D & Q Truy Vấn O(1) 

// Prefix Sum 1D chuẩn 1-based index, ép kiểu long long chống tràn số
#include <iostream>
#include <vector>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, q;
    if (!(cin >> n >> q)) return 0;

    vector<long long> a(n + 1), p(n + 1, 0);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        p[i] = p[i - 1] + a[i]; // Tiền xử lý O(N)
    }

    // Trả lời Q truy vấn O(1)
    while (q--) {
        int l, r;
        cin >> l >> r;
        long long sum = p[r] - p[l - 1];
        cout << sum << "
";
    }
    return 0;
}

Code Mẫu 2: Prefix Sum 2D (Ma Trận Con) 

// Prefix Sum 2D tính tổng ma trận con trong O(1)
#include <iostream>
#include <vector>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, m, q;
    if (!(cin >> n >> m >> q)) return 0;

    vector<vector<long long>> p(n + 1, vector<long long>(m + 1, 0));
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            long long val; cin >> val;
            p[i][j] = p[i - 1][j] + p[i][j - 1] - p[i - 1][j - 1] + val;
        }
    }

    while (q--) {
        int x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;
        long long ans = p[x2][y2] - p[x1 - 1][y2] - p[x2][y1 - 1] + p[x1 - 1][y1 - 1];
        cout << ans << "
";
    }
    return 0;
}

Bản Chất Mảng Hiệu (Difference Array)

Mảng hiệu $D$ được định nghĩa: $D[i] = A[i] - A[i - 1]$ (với $D[1] = A[1]$).
Khi cần cộng thêm giá trị $V$ vào tất cả các phần tử trong đoạn $[L, R]$, thay vì dùng vòng for $O(N)$, ta chỉ cần tác động vào đúng 2 vị trí biên trong $O(1)$:

D[L] += V;      D[R + 1] -= V;

Ý nghĩa sư phạm: Sau khi hoàn tất tất cả $Q$ thao tác cập nhật, ta tính mảng cộng dồn trên $D$ để khôi phục toàn bộ mảng $A$ cuối cùng chỉ trong đúng 1 lần duyệt $O(N)$!
Tổng chi phí thuật toán: $O(Q + N)$ thay vì $O(Q imes N)$!

int main() {

    int n, q;

// Sắp xếp sự kiện CSES Restaurant Customers - O(N log N)
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct Event {
    int time;
    int type; // +1: Đến, -1: Rời đi
};

bool cmp(const Event &a, const Event &b) {
    if (a.time != b.time) return a.time < b.time;
    return a.type < b.type; // -1 đứng trước +1 nếu cùng thời điểm
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n;
    if (!(cin >> n)) return 0;

    vector<Event> events;
    for (int i = 0; i < n; i++) {
        int a, b; cin >> a >> b;
        events.push_back({a, +1});
        events.push_back({b, -1});
    }

    sort(events.begin(), events.end(), cmp);

    int maxCust = 0, curCust = 0;
    for (const auto &e : events) {
        curCust += e.type;
        maxCust = max(maxCust, curCust);
    }

    cout << maxCust << "
";
    return 0;
}

Phép Tính Modulo & Lũy Thừa Nhị Phân

Cộng: (a + b) % MOD
Trừ: (a - b % MOD + MOD) % MOD
Nhân: (1LL * (a % MOD) * (b % MOD)) % MOD
long long binpow(long long a, long long b, long long m) {
    long long res = 1; a %= m;
    while (b > 0) {
        if (b & 1) res = (1LL * res * a) % m;
        a = (1LL * a * a) % m;
        b >>= 1;
    }
    return res;
}
Số Học • Ước Số O(log(min))

long long gcd(long long a, long long b) {

long long lcm(long long a, long long b) {

    return (a / __gcd(a, b)) * b;
