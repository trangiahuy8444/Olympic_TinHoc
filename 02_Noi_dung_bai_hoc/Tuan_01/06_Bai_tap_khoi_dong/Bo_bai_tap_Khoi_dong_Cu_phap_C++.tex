\section*{\color{AsianDarkBlue}Bài 1: Tính toán hình học cơ bản}
\textbf{Đề bài:} Nhập vào 2 số nguyên dương $a, b$ ($1 \le a, b \le 10^6$) là chiều dài và chiều rộng của một hình chữ nhật.
\begin{itemize}
    \setlength{\itemsep}{1pt}
    \item Dòng 1 in ra chu vi hình chữ nhật: $P = (a + b) \times 2$.
    \item Dòng 2 in ra diện tích hình chữ nhật: $S = a \times b$.
\end{itemize}
\textbf{Lưu ý:} Biến diện tích có thể lên tới $10^{12}$, bắt buộc dùng kiểu \texttt{long long}.\\
\textbf{Bảng testcase mẫu:}
\begin{table}[h!]
\centering\small
\begin{tabular}{|p{7.5cm}|p{7.5cm}|}
\hline
\textbf{Input (Bàn phím)} & \textbf{Output (Màn hình)} \\ \hline
\texttt{100000 200000} & \texttt{600000} \\
& \texttt{20000000000} \\ \hline
\end{tabular}
\end{table}

\section*{\color{AsianDarkBlue}Bài 2: Tìm số lớn nhất trong ba số (Rèn luyện if/else)}
\textbf{Đề bài:} Nhập vào 3 số nguyên $a, b, c$ ($|a|, |b|, |c| \le 10^9$). Hãy in ra giá trị lớn nhất trong 3 số.\\
\textbf{Gợi ý:} Khởi tạo biến \texttt{max\_val = a}, sau đó so sánh lần lượt với $b$ và $c$.\\
\textbf{Ví dụ:} Input \texttt{15 -7 28} $\implies$ Output là \texttt{28}.

\newpage

\section*{\color{AsianDarkBlue}Bài 3: Kiểm tra năm nhuận (Toán tử Logic \&\& và ||)}
\textbf{Đề bài:} Nhập vào một năm nguyên dương $N$ ($1 \le N \le 10^5$). Một năm là năm nhuận nếu:
\begin{itemize}
    \setlength{\itemsep}{1pt}
    \item Năm đó chia hết cho 400, HOẶC:
    \item Năm đó chia hết cho 4 nhưng KHÔNG chia hết cho 100.
\end{itemize}
In ra \texttt{YES} nếu là năm nhuận, ngược lại in \texttt{NO}.\\
\textbf{Bảng testcase mẫu:}
\begin{table}[h!]
\centering\small
\begin{tabular}{|p{7.5cm}|p{7.5cm}|}
\hline
\textbf{Input} & \textbf{Output} \\ \hline
\texttt{2024} & \texttt{YES} \\ \hline
\texttt{1900} & \texttt{NO} \\ \hline
\end{tabular}
\end{table}

\section*{\color{AsianDarkBlue}Bài 4: Tính tổng các số chẵn từ 1 đến N (Vòng lặp for)}
\textbf{Đề bài:} Nhập vào số nguyên dương $N$ ($1 \le N \le 10^6$). Hãy tính tổng tất cả các số chẵn trong khoảng từ 1 đến $N$.\\
\textbf{Lưu ý:} Biến tổng tích lũy phải dùng kiểu \texttt{long long} và khởi tạo bằng 0 (\texttt{long long tong = 0;}).\\
\textbf{Ví dụ:} Với $N = 10$, các số chẵn là $2, 4, 6, 8, 10 \implies$ Tổng là \texttt{30}.

\section*{\color{AsianDarkBlue}Bài 5: Tập đọc ghi tệp phòng thi Olympic (Tổng dãy số)}
\textbf{Quy cách tệp:} Đọc dữ liệu từ tệp \texttt{TONG.INP}, ghi kết quả ra tệp \texttt{TONG.OUT}.\\
\textbf{Hướng dẫn:} Đặt 2 dòng code sau vào đầu hàm \texttt{main()}:
\begin{center}
\texttt{freopen("TONG.INP", "r", stdin); freopen("TONG.OUT", "w", stdout);}
\end{center}
\textbf{Đề bài:} 
\begin{itemize}
    \setlength{\itemsep}{1pt}
    \item Dòng đầu chứa số nguyên dương $N$ ($1 \le N \le 10^5$).
    \item Dòng thứ hai chứa $N$ số nguyên $A_1, A_2, \dots, A_N$ ($|A_i| \le 10^9$).
\end{itemize}
Hãy tính và in ra tổng của $N$ số nguyên đó.\\
\textbf{Bảng testcase mẫu:}
\begin{table}[h!]
\centering\small
\begin{tabular}{|p{7.5cm}|p{7.5cm}|}
\hline
\textbf{Tệp TONG.INP} & \textbf{Tệp TONG.OUT} \\ \hline
\texttt{4} & \texttt{125} \\
\texttt{10 20 -5 100} & \\ \hline
\end{tabular}
\end{table}
