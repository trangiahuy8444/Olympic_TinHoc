\section*{\color{AsianDarkBlue}\normalsize 2. Thống nhất Mục tiêu \& Quy định Đội tuyển}
\begin{enumerate}
    \setlength{\itemsep}{2pt}
    \small
    \item \textbf{Mục tiêu điểm số:} Đạt tối thiểu \textbf{12.0 -- 15.0 / 20.0 điểm} (Ăn trọn Bài 1, ăn chắc subtask Bài 2 và Bài 3).
    \item \textbf{Kỷ luật thời gian:} Có mặt đúng giờ các buổi học trực tiếp để trao đổi tư duy giải thuật và sửa code tối ưu.
    \item \textbf{Làm bài tập rèn luyện tại nhà (Bắt buộc):} Chủ động sắp xếp thời gian linh hoạt trong tuần để hoàn thành đầy đủ phiếu bài tập được giao (khoảng 3--5 bài). Điều cốt lõi là tinh thần tự giác, tự tay viết code và nộp bài trước buổi học tiếp theo (không áp đặt gò bó số giờ cụ thể).
    \item \textbf{Quy tắc vượt khó khi gặp bài khó:} Suy nghĩ và nháp tối thiểu 20--30 phút $\to$ Viết code vét cạn trước $\to$ Trao đổi trên nhóm học tập cùng thầy/cô và bạn bè; hiểu rõ bản chất thuật toán.
    \item \textbf{Tâm thế làm quen C++:} Không ngại sai cú pháp (dấu `;`, kiểu dữ liệu). Sau 2--3 tuần thực hành liên tục, việc viết C++ sẽ trở nên tự nhiên và thuần thục.
\end{enumerate}

\section*{\color{AsianDarkBlue}\normalsize 3. Hướng dẫn thiết lập môi trường C++ trên Code::Blocks từng bước}

\begin{tcolorbox}[colback=AsianLightBlue,colframe=AsianBlue,arc=1mm,boxrule=0.5pt]
\footnotesize\textbf{Tại sao chọn Code::Blocks?} Code::Blocks là IDE tiêu chuẩn được cài đặt mặc định trên các máy thi Học sinh giỏi các cấp (Quận/Huyện, TP.HCM, Quốc gia). Bản cài tích hợp sẵn MinGW (GCC/G++), phím tắt biên dịch nhanh \texttt{F9}, không đòi hỏi cấu hình phức tạp.
\end{tcolorbox}

\subsection*{\color{AsianBlue}\small Bước 1: Tải và cài đặt Code::Blocks (Bản tích hợp MinGW)}
\begin{itemize}
    \setlength{\itemsep}{1.5pt}
    \small
    \item Tải bộ cài tại: \url{https://www.codeblocks.org/downloads/binaries/}.
    \item \textbf{Đặc biệt lưu ý:} Chọn bản \textbf{\texttt{codeblocks-20.03mingw-setup.exe}} (bản có chữ \texttt{mingw} chứa sẵn trình biên dịch GCC).
    \item Cài đặt theo thiết lập mặc định. Khi mở lên lần đầu, chọn \textbf{GNU GCC Compiler} $\to$ \textbf{Set as default} $\to$ \textbf{OK}.
\end{itemize}

\subsection*{\color{AsianBlue}\small Bước 2: Cấu hình chuẩn C++17 và Phím tắt vàng}
\begin{itemize}
    \setlength{\itemsep}{1.5pt}
    \small
    \item \textbf{Cấu hình C++17:} Vào menu \textbf{Settings} $\to$ \textbf{Compiler...} $\to$ Tab \textbf{Compiler Flags} $\to$ Tích chọn \texttt{Have g++ follow the C++17 ISO C++ language standard [-std=c++17]} và \texttt{Optimize generated code [-O2]} $\to$ Bấm \textbf{OK}.
    \item \textbf{Phím tắt vàng:} \texttt{Ctrl + Shift + N} (tạo file mới), \texttt{Ctrl + S} (lưu file đuôi \texttt{.cpp}), \textbf{\texttt{F9}} (\textbf{Build and Run} -- Dịch và chạy ngay lập tức!).
\end{itemize}
