\section*{\color{AsianDarkBlue}Bài 1: Viết hàm tính lũy thừa \& Kiểm tra số nguyên tố}
\textbf{Đề bài:} Viết chương trình gồm 2 hàm độc lập:
\begin{enumerate}
    \setlength{\itemsep}{1pt}
    \item Hàm \texttt{long long power(long long a, int b)} tính $a^b$ ($a \ge 1, 0 \le b \le 18, a^b \le 10^{18}$).
    \item Hàm \texttt{bool is\_prime(long long n)} kiểm tra số nguyên tố trong độ phức tạp $O(\sqrt{n})$.
\end{enumerate}
Trong hàm \texttt{main()}, nhập số nguyên dương $x$ ($1 \le x \le 10^{12}$). In ra \texttt{YES} nếu $x$ là số nguyên tố, ngược lại in \texttt{NO}.\\
\textbf{Bảng testcase mẫu:}
\begin{table}[h!]
\centering\small
\begin{tabular}{|p{7.5cm}|p{7.5cm}|}
\hline
\textbf{Input (Bàn phím)} & \textbf{Output (Màn hình)} \\ \hline
\texttt{999999999989} & \texttt{YES} \\ \hline
\texttt{1000000000000} & \texttt{NO} \\ \hline
\end{tabular}
\end{table}

\section*{\color{AsianDarkBlue}Bài 2: Viết hàm tự hoán đổi \& Kiểm chứng tham chiếu \texttt{\&}}
\textbf{Đề bài:} Viết hàm \texttt{void my\_swap(long long \&u, long long \&v)} hoán đổi giá trị 2 biến. Nhập vào 2 số nguyên $A, B$ ($|A|, |B| \le 10^{18}$). Gọi hàm \texttt{my\_swap(A, B)} và in ra giá trị của $A, B$ sau khi hoán đổi.\\
\textbf{Lưu ý sư phạm:} Hãy thử bỏ dấu \texttt{\&} ở tham số để thấy giá trị $A, B$ ngoài \texttt{main()} không hề thay đổi!\\
\textbf{Bảng testcase mẫu:}
\begin{table}[h!]
\centering\small
\begin{tabular}{|p{7.5cm}|p{7.5cm}|}
\hline
\textbf{Input} & \textbf{Output} \\ \hline
\texttt{100 200} & \texttt{200 100} \\ \hline
\end{tabular}
\end{table}

\section*{\color{AsianDarkBlue}Bài 3: Cài đặt Mảng đếm tần số trực tiếp $O(N)$}
\textbf{Đề bài:} Cho mảng $A$ gồm $N$ số nguyên không âm ($1 \le N \le 10^5, 0 \le A_i \le 10^6$). Hãy đếm số lần xuất hiện của từng giá trị trong mảng và in ra theo định dạng: \texttt{Giá\_trị: Số\_lần} theo thứ tự giá trị tăng dần (chỉ in các số có xuất hiện ít nhất 1 lần).\\
\textbf{Khung thuật toán chuẩn phòng thi:}
\begin{lstlisting}[language=C++]
const int MAX_VAL = 1000005;
int cnt[MAX_VAL]; // Mang dem toan cuc tren vung BSS (tu dong = 0)

for (int i = 0; i < n; i++) {
    int x; cin >> x;
    cnt[x]++; // Chi so la gia tri, noi dung la so lan xuat hien
}
\end{lstlisting}
\textbf{Bảng testcase mẫu:}
\begin{table}[h!]
\centering\small
\begin{tabular}{|p{7.5cm}|p{7.5cm}|}
\hline
\textbf{Input} & \textbf{Output} \\ \hline
\texttt{7} & \texttt{2: 1} \\
\texttt{5 2 8 5 9 2 5} & \texttt{5: 3} \\
& \texttt{8: 1} \\
& \texttt{9: 1} \\ \hline
\end{tabular}
\end{table}

\section*{\color{AsianDarkBlue}Bài 4: Thực hành tệp Olympic: Đếm phần tử phân biệt (UNIQUE)}
\textbf{Quy cách tệp phòng thi:} Đọc từ tệp \texttt{UNIQUE.INP}, ghi kết quả ra tệp \texttt{UNIQUE.OUT}.\\
\textbf{Đề bài:} Cho dãy gồm $N$ số nguyên ($1 \le N \le 10^5, 0 \le A_i \le 10^6$). Hãy đếm xem trong dãy có bao nhiêu giá trị phân biệt (khác nhau đôi một).\\
\textbf{Độ phức tạp yêu cầu:} $O(N)$ thời gian, $O(\max(A_i))$ bộ nhớ. Tuyệt đối không dùng 2 vòng for $O(N^2)$ vì sẽ bị loại TLE ngay lập tức!\\

\hline
\textbf{Tệp UNIQUE.INP} & \textbf{Tệp UNIQUE.OUT} \\ \hline
\texttt{6} & \texttt{4} \\
\texttt{10 20 10 30 20 50} & \\ \hline
\end{tabular}
\end{table}
