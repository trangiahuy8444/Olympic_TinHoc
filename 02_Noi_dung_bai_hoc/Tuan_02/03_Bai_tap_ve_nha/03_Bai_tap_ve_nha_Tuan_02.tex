\section*{\color{AsianDarkBlue}\small PHẦN A: THAO TÁC CĂN BẢN TRÊN MẢNG (MỨC 1)}
\vspace{-0.2cm}

\textbf{\color{AsianBlue}Bài 1: Liệt kê các phần tử chẵn.} Nhập $N$ ($1 \le N \le 10^5$) và mảng $A_1, \dots, A_N$ ($|A_i| \le 10^9$). In tất cả các số chẵn theo thứ tự xuất hiện (cách nhau dấu cách). Nếu không có số chẵn nào, in \texttt{-1}.\\
\textit{Ví dụ:} Input \texttt{6} / \texttt{1 4 7 8 10 3} $\to$ Output: \texttt{4 8 10}.

\vspace{0.1cm}
\noindent
\textbf{\color{AsianBlue}Bài 2: Tìm Max, Min và độ chênh lệch.} Cho mảng $A$ gồm $N$ số nguyên ($1 \le N \le 10^5$, $|A_i| \le 10^9$). Tìm giá trị lớn nhất ($Max$), nhỏ nhất ($Min$) và hiệu số $(Max - Min)$.\\
\textit{Ví dụ:} Input \texttt{5} / \texttt{-3 7 2 15 -8} $\to$ Output: \texttt{15 -8 23}.\\
\textit{Lưu ý:} $Max - Min$ có thể chạm $2 \cdot 10^9 \implies$ Dùng kiểu \texttt{long long} và khởi tạo $Max, Min$ an toàn với \texttt{A[0]}.

\vspace{0.1cm}
\noindent
\textbf{\color{AsianBlue}Bài 3: Tổng phần tử đầu, cuối và trung tâm.} Cho số nguyên dương lẻ $N$ ($3 \le N \le 10^5$) và mảng $A$. Tính tổng của 3 phần tử: phần tử đầu tiên ($A_1$), phần tử chính giữa ($A_{(N+1)/2}$), và phần tử cuối cùng ($A_N$).\\
\textit{Ví dụ:} Input \texttt{5} / \texttt{10 20 30 40 50} $\to$ Output: \texttt{90} ($10 + 30 + 50$).

\vspace{0.1cm}
\noindent
\textbf{\color{AsianBlue}Bài 4: Đếm và tính tổng các số dương.} Cho mảng $A$ gồm $N$ số nguyên ($1 \le N \le 10^5$, $|A_i| \le 10^9$). Đếm số lượng phần tử dương ($A_i > 0$) và tính tổng các phần tử đó.\\
\textit{Ví dụ:} Input \texttt{6} / \texttt{-2 5 0 8 -4 10} $\to$ Output: \texttt{3 23} (các số dương: 5, 8, 10; tổng: 23).

\vspace{-0.2cm}
\section*{\color{AsianDarkBlue}\small PHẦN B: DUYỆT MẢNG, ĐẾM TẦN SỐ \& CHUẨN TỆP (MỨC 2)}
\vspace{-0.2cm}

\textbf{\color{AsianBlue}Bài 5: Đếm số lần xuất hiện của K.} Cho mảng $A$ ($N$ số) và số nguyên $K$ ($1 \le N \le 10^5$, $|A_i|, |K| \le 10^9$). Đếm số lần $K$ xuất hiện và in vị trí (1-based) xuất hiện đầu tiên của $K$ (nếu không có, vị trí là \texttt{-1}).\\
\textit{Ví dụ:} Input \texttt{7 3} / \texttt{1 3 5 3 7 3 9} $\to$ Dòng 1: \texttt{3}, Dòng 2: \texttt{2}.

\newpage

\textbf{\color{AsianBlue}Bài 6: Đảo ngược mảng (Chuẩn tệp DAO.INP / DAO.OUT).} Đọc mảng $N$ phần tử từ tệp \texttt{DAO.INP} ($1 \le N \le 10^5$, $|A_i| \le 10^9$), in mảng theo thứ tự đảo ngược từ cuối về đầu ra tệp \texttt{DAO.OUT}.\\
\textit{Ví dụ:} \texttt{DAO.INP}: \texttt{4} / \texttt{1 2 3 4} $\to$ \texttt{DAO.OUT}: \texttt{4 3 2 1}.\\
\textit{Cú pháp chuẩn:}
\begin{lstlisting}[language=C++]
freopen("DAO.INP", "r", stdin);
freopen("DAO.OUT", "w", stdout);
\end{lstlisting}

\vspace{0.15cm}
\noindent
\textbf{\color{AsianBlue}Bài 7: Kiểm tra dãy số tăng dần.} Cho mảng $A$ ($2 \le N \le 10^5$). Kiểm tra dãy có tăng ngặt không ($A_i < A_{i+1}$ với mọi $i$). In \texttt{YES} nếu tăng ngặt, ngược lại in \texttt{NO}.\\
\textit{Ví dụ:} Input \texttt{5} / \texttt{1 3 7 9 12} $\to$ \texttt{YES}; Input \texttt{4} / \texttt{2 5 5 9} $\to$ \texttt{NO}.

\vspace{0.15cm}
\noindent
\textbf{\color{AsianBlue}Bài 8: Đếm số nguyên tố trong mảng.} Cho mảng $A$ gồm $N$ số nguyên dương ($1 \le N \le 10^4$, $1 \le A_i \le 10^6$). Đếm số lượng các số nguyên tố và tính tổng của chúng.\\
\textit{Ví dụ:} Input \texttt{6} / \texttt{1 2 3 4 5 6} $\to$ Dòng 1: \texttt{3}, Dòng 2: \texttt{10} (các số nguyên tố: 2, 3, 5; số 1 không phải số nguyên tố).

\vspace{-0.2cm}
\section*{\color{AsianDarkBlue}\small PHẦN C: VÒNG LẶP LỒNG NHAU \& THỬ THÁCH OLYMPIC (MỨC 3)}
\vspace{-0.2cm}

\textbf{\color{AsianBlue}Bài 9: Đếm số cặp có tổng bằng S.} Cho mảng $A$ gồm $N$ số nguyên và số $S$ ($2 \le N \le 2000$, $|A_i|, |S| \le 10^9$). Đếm số lượng cặp chỉ số $(i, j)$ với $1 \le i < j \le N$ sao cho $A_i + A_j = S$.\\
\textit{Ví dụ:} Input \texttt{5 8} / \texttt{3 5 2 6 5} $\to$ Output: \texttt{3} (gồm các cặp $(1,2), (1,5)$ cho $3+5=8$, và $(3,4)$ cho $2+6=8$).\\
\textit{Tư duy:} Dùng 2 vòng lặp $O(N^2)$: $N = 2000 \implies \approx 2 \cdot 10^6$ phép tính, chạy $\approx 0.01$s. Chú ý ép kiểu \texttt{1LL * A[i] + A[j] == S}.

\vspace{0.15cm}
\noindent
\textbf{\color{AsianBlue}Bài 10: Đoạn con liên tiếp có tổng lớn nhất (Thuật toán Kadane).} Cho mảng $A$ gồm $N$ số ($1 \le N \le 10^5$, $|A_i| \le 10^6$). Tìm đoạn con liên tiếp $A_i, A_{i+1}, \dots, A_j$ ($1 \le i \le j \le N$) có tổng lớn nhất.\\
\textit{Ví dụ:} Input \texttt{8} / \texttt{-2 -3 4 -1 -2 1 5 -3} $\to$ Output: \texttt{7} (đoạn con $[4, -1, -2, 1, 5]$ có tổng $7$).\\
\textit{Mã nguồn mẫu tối ưu $O(N)$:}
\begin{lstlisting}[language=C++]
long long max_sum = a[0], cur_sum = 0;
for (int i = 0; i < n; i++) {
    cur_sum += a[i];
    max_sum = max(max_sum, cur_sum);
    if (cur_sum < 0) cur_sum = 0;
}
cout << max_sum << "\n";
\end{lstlisting}
