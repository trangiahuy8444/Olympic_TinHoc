\section*{\color{AsianDarkBlue}Bài 1: Nhập xuất mảng \& Tính trung bình cộng số dương}
\textbf{Đề bài:} Cho mảng $A$ gồm $N$ số nguyên ($1 \le N \le 10^5, |A_i| \le 10^9$). Hãy tính và in ra trung bình cộng của các số dương ($A_i > 0$) trong mảng, định dạng lấy 2 chữ số thập phân sau dấu phẩy. Nếu mảng không có số dương nào, in ra \texttt{-1}.\\
\textbf{Gợi ý kỹ thuật:} Biến tổng các số dương dùng kiểu \texttt{long long}, biến đếm số lượng dương kiểu \texttt{int}. Khi chia phải ép kiểu: \texttt{(double)tong / dem}.\\

\centering\small
\begin{tabular}{|p{7.5cm}|p{7.5cm}|}
\hline
\textbf{Input (Bàn phím)} & \textbf{Output (Màn hình)} \\ \hline
\texttt{5} & \texttt{6.00} \\
\texttt{-3 4 8 -1 6} & \\ \hline
\end{tabular}
\end{table}

\section*{\color{AsianDarkBlue}Bài 2: Tìm Max, Min và vị trí đầu tiên xuất hiện}
\textbf{Đề bài:} Cho mảng $A$ gồm $N$ số nguyên ($1 \le N \le 10^5, |A_i| \le 10^9$). Hãy tìm giá trị lớn nhất ($Max$), vị trí chỉ số $1$-based đầu tiên của $Max$, giá trị nhỏ nhất ($Min$) và vị trí $1$-based đầu tiên của $Min$.\\

\section*{\color{AsianDarkBlue}Bài 3: Đảo ngược mảng tại chỗ (Two Pointers In-place)}
\textbf{Đề bài:} Cho mảng $A$ gồm $N$ phần tử ($1 \le N \le 10^5$). Hãy đảo ngược thứ tự các phần tử của mảng $A$ mà \textbf{không dùng thêm bất kỳ mảng phụ nào} (tối ưu bộ nhớ $O(1)$ phụ trợ).\\
\textbf{Khung thuật toán chuẩn (Two Pointers):}
\begin{lstlisting}[language=C++]
// Dao nguoc mang bang cach hoan doi doi xung
for (int i = 0; i < n / 2; i++) {
    swap(a[i], a[n - 1 - i]);
}
\end{lstlisting}

\centering\small
\begin{tabular}{|p{7.5cm}|p{7.5cm}|}
\hline
\textbf{Input} & \textbf{Output} \\ \hline
\texttt{5} & \texttt{10 8 6 4 2} \\
\texttt{2 4 6 8 10} & \\ \hline
\end{tabular}
\end{table}

\section*{\color{AsianDarkBlue}Bài 4: Thực hành đọc xuất tệp chuẩn Olympic (DAO.INP / DAO.OUT)}
\textbf{Quy cách tệp phòng thi:}
\begin{itemize}
    \setlength{\itemsep}{1pt}
    \item Tệp đầu vào: \texttt{DAO.INP}
    \item Tệp kết quả: \texttt{DAO.OUT}
\end{itemize}
\textbf{Đề bài:} Đọc từ tệp \texttt{DAO.INP} gồm số nguyên $N$ ($1 \le N \le 10^5$) và $N$ số nguyên của mảng $A$. Đảo ngược toàn bộ mảng và ghi ra tệp \texttt{DAO.OUT} trên một dòng duy nhất, mỗi phần tử cách nhau một dấu cách.\\
\textbf{Hướng dẫn thao tác trên Code::Blocks:}
\begin{enumerate}
    \setlength{\itemsep}{1pt}
    \item Mở Code::Blocks, nhấn \texttt{Ctrl+Shift+N} tạo tệp \texttt{DAO.cpp} và lưu vào thư mục bài làm.
    \item Tạo tệp văn bản \texttt{DAO.INP} nằm cùng thư mục với file \texttt{DAO.cpp}.
    \item Dùng cặp lệnh \texttt{freopen} chuyển hướng luồng và nhấn \texttt{F9} chạy chương trình.
\end{enumerate}
