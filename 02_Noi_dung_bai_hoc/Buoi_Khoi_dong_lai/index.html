SUMSEQ — Tính Tổng Dãy Số Lớn ($N \le 10^9$)

Tệp: SUMSEQ.INP / OUT

Cho ba số nguyên dương $A, B, D$ ($A \le B \le 10^9, D \le 10^5$). Dãy số cấp số cộng gồm các phần tử: $A, A + D, A + 2D, \dots \le B$. Tính tổng tất cả phần tử không vượt quá $B$.

SUMSEQ.INP: 3 15 4
SUMSEQ.OUT: 36
Thuật toán $O(1)$ tránh TLE & Tràn số: 
#include <iostream>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    long long A, B, D;
    if (!(cin >> A >> B >> D)) return 0;

    if (A > B) { cout << 0 << "\n"; return 0; }

    long long n = (B - A) / D + 1;
    long long last = A + (n - 1) * D;

    // Tinh n * (A + last) / 2 bang cach chia truoc de chong tran so
    long long ans;
    if (n % 2 == 0) ans = (n / 2) * (A + last);
    else ans = n * ((A + last) / 2);

    cout << ans << "\n";
    return 0;
}

PAIRPROD — Tích Lớn Nhất Của 2 Số ($N \le 10^5, A_i \le 10^9$)

Tệp: PAIRPROD.INP / OUT

Cho mảng $N$ số nguyên dương. Tìm tích lớn nhất của 2 phần tử bất kỳ. Chú ý: $A_i$ lên tới $10^9$, tích có thể đạt $10^{18}$, nếu dùng kiểu int sẽ bị âm! 

#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n;
    if (!(cin >> n)) return 0;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    sort(a.begin(), a.end());
    // Tich lon nhat cua 2 so duong lon nhat o cuoi mang
    long long ans = a[n - 1] * a[n - 2];
    cout << ans << "\n";
    return 0;
}

Mảng Đếm Tần Số (Counting Array) & Kỹ Thuật Xử Lý Xâu

Bẻ gãy độ phức tạp từ $O(N^2)$ xuống $O(N)$ bằng mảng đánh dấu tần số trực tiếp; làm chủ thư viện <string> và bảng mã ASCII trong các bài toán chuẩn hóa xâu.

Bản Chất Mảng Đếm Tần Số $O(N)$

int cnt[MAX] = {0}; // Khai bao toan cuc BSS

// Nap gia tri vao mang dem: O(1)

// Kiem tra gia tri co ton tai khong: O(1)

Lưu ý bộ nhớ: Khai báo mảng tĩnh toàn cục có thể chứa tới $10^7$ số nguyên (khoảng 40MB RAM), thoải mái trong giới hạn 256MB của kỳ thi.

PAIRSUM — Đếm Cặp Tổng Bằng K ($N \le 10^5, A_i \le 10^6$)

PAIRSUM.INP / OUT

Cho mảng $N$ số nguyên không âm và số $K$. Đếm số cặp $(i, j)$ với $i < j$ sao cho $A_i + A_j = K$. 

#include <iostream>
#include <vector>
using namespace std;

const int MAX = 2000005;
int cnt[MAX] = {0};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, k;
    if (!(cin >> n >> k)) return 0;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    long long ans = 0;
    for (int i = 0; i < n; i++) {
        int target = k - a[i];
        if (target >= 0 && target < MAX) {
            ans += cnt[target]; // Da gap bao nhieu so thoa man truoc do
        }
        if (a[i] >= 0 && a[i] < MAX) {
            cnt[a[i]]++;
        }
    }

    cout << ans << "\n";
    return 0;
}
Bài 2 • Xử lý xâu

PALIN — Xâu Đối Xứng Nâng Cao (Bỏ qua ký tự đặc biệt)

PALIN.INP / OUT

Kiểm tra xem xâu có đối xứng không sau khi đã loại bỏ dấu cách, dấu câu và không phân biệt hoa thường. Ví dụ: "A man, a plan, a canal: Panama!" $\to$ YES. 

#include <iostream>
#include <string>
#include <cctype>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    string s;
    if (!getline(cin, s)) return 0;

    string clean = "";
    for (char c : s) {
        if (isalnum(c)) clean += tolower(c);
    }

    int left = 0, right = (int)clean.length() - 1;
    bool ok = true;
    while (left < right) {
        if (clean[left] != clean[right]) { ok = false; break; }
        left++; right--;
    }

    if (ok) cout << "YES\n";
    else cout << "NO\n";

    return 0;
}

Kỹ Thuật Tiền Xử Lý: Mảng Cộng Dồn (Prefix Sum 1D & 2D)

Tính tổng đoạn $[L, R]$ trong $O(1)$ thay vì duyệt lặp $O(N)$ bằng công thức $Pref[R] - Pref[L-1]$; ứng dụng xử lý hàng triệu truy vấn tổng chỉ trong chớp mắt.

Công Thức Cốt Lõi Prefix Sum

Khởi tạo (1-based index):

Pref[0] = 0 Pref[i] = Pref[i - 1] + A[i] (1 ≤ i ≤ N)

Truy vấn trong $O(1)$:

Sum(L, R) = Pref[R] - Pref[L - 1]

Nếu có $Q = 10^5$ truy vấn: Duyệt thông thường mất $O(N \cdot Q) = 10^{10}$ phép tính (chắc chắn dính TLE). Dùng Prefix Sum chỉ mất $O(N + Q) \approx 2 \cdot 10^5$ phép tính (chạy dưới 0.05 giây!).

Mở Rộng: Prefix Sum 2D Trên Bảng

Tính tổng hình chữ nhật từ $(x_1, y_1)$ đến $(x_2, y_2)$ trong $O(1)$:

Pref[x][y] = A[x][y] + Pref[x-1][y] + Pref[x][y-1] - Pref[x-1][y-1];

Sum = Pref[x2][y2] - Pref[x1-1][y2] - Pref[x2][y1-1] + Pref[x1-1][y1-1];

RANGESUM — Q Truy Vấn Tổng Đoạn ($N, Q \le 10^5$)

RANGESUM.INP / OUT 
#include <iostream>
#include <vector>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, q;
    if (!(cin >> n >> q)) return 0;

    vector<long long> pref(n + 1, 0);
    for (int i = 1; i <= n; i++) {
        long long val;
        cin >> val;
        pref[i] = pref[i - 1] + val;
    }

    while (q--) {
        int l, r;
        cin >> l >> r;
        cout << pref[r] - pref[l - 1] << "\n";
    }

    return 0;
}

Mảng Hiệu (Difference Array) & Sắp Xếp Tùy Biến

Tăng giảm đoạn $[L, R]$ thêm $V$ đơn vị trong $O(1)$ thay vì $O(N)$; áp dụng giải quyết triệt để các bài toán đặt phòng, phục vụ quán ăn và chồng lấn thời gian.

Kỹ Thuật Mảng Hiệu (Difference Array)

Khi cần thực hiện $Q$ lần thao tác: "Cộng thêm $V$ vào tất cả phần tử từ chỉ số $L$ đến $R$":

// 1. Cap nhat trong O(1):
diff[L] += V;
diff[R + 1] -= V;
// 2. Khoi phuc mang sau Q lan cap nhat: O(N)
for (int i = 1; i <= n; i++) cur += diff[i]; a[i] = cur;

Tổng độ phức tạp là $O(Q + N)$ thay vì $O(Q \cdot N)$, cứu nguy ngoạn mục cho bài toán cập nhật đoạn.

Sắp Xếp Tùy Biến (Custom Comparator) 

struct ThiSinh {
    string ten;
    int diem;
};

// Tra ve true neu a dung truoc b
bool cmp(const ThiSinh &a, const ThiSinh &b) {
    if (a.diem != b.diem) return a.diem > b.diem; // Diem cao xep truoc
    return a.ten < b.ten; // Bang diem: xep theo ten Alphabet
}

sort(ds.begin(), ds.end(), cmp);

RESTAURANT — Quản Lý Đặt Bàn Nhà Hàng ($N \le 10^5$)

RESTAURANT.INP / OUT

Có $N$ đoàn khách đặt bàn, đoàn thứ $i$ ăn từ mốc $L_i$ đến $R_i$. Hỏi nhà hàng cần tối thiểu bao nhiêu bàn để phục vụ tất cả khách không bị trùng? 

#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAX_TIME = 1000005;
int diff_tbl[MAX_TIME] = {0};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n;
    if (!(cin >> n)) return 0;

    int max_r = 0;
    for (int i = 0; i < n; i++) {
        int l, r;
        cin >> l >> r;
        diff_tbl[l] += 1;
        diff_tbl[r + 1] -= 1;
        max_r = max(max_r, r);
    }

    int cur_tables = 0, ans = 0;
    for (int t = 1; t <= max_r; t++) {
        cur_tables += diff_tbl[t];
        ans = max(ans, cur_tables);
    }

    cout << ans << "\n";
    return 0;
}

Số Học • Dãy Số O(1)

Tổng Dãy Số & Cấp Số Cộng

1 + 2 + ... + n = n * (n + 1) / 2
Số số hạng: n = (un - u1) / d + 1
Tổng cấp số cộng: S = n * (u1 + un) / 2
long long n = (B - A) / D + 1;
long long last = A + (n - 1) * D;
long long ans = (n % 2 == 0) ? (n / 2) * (A + last) : n * ((A + last) / 2);

⚠️ Bẫy: Chia 2 trước nếu n chẵn để tránh tràn số tích n * (A + last).

Số Học • Modulo O(log B)

Phép Tính Modulo & Lũy Thừa Nhị Phân

Cộng: (a + b) % MOD
Trừ: (a - b % MOD + MOD) % MOD
Nhân: (1LL * (a % MOD) * (b % MOD)) % MOD
long long binpow(long long a, long long b, long long m) {
    long long res = 1; a %= m;
    while (b > 0) {
        if (b & 1) res = (1LL * res * a) % m;
        a = (1LL * a * a) % m;
        b >>= 1;
    }
    return res;
}
Số Học • Ước Số O(log(min))

Ước Chung Lớn Nhất & Bội Chung Nhỏ Nhất

Hàm có sẵn: __gcd(a, b) trong <algorithm>. Công thức liên hệ: a * b = gcd(a, b) * lcm(a, b). 

long long gcd(long long a, long long b) {
    return b == 0 ? a : gcd(b, a % b);
}

long long lcm(long long a, long long b) {
    // Chia truoc de chong tran so a * b!
    return (a / __gcd(a, b)) * b;
}
Số Học • Nguyên Tố O(N log log N)

Sàng Số Nguyên Tố Eratosthenes ($N \le 10^7$)

const int MAX = 10000005;
bool is_prime[MAX];

void sieve() {
    fill(is_prime, is_prime + MAX, true);
    is_prime[0] = is_prime[1] = false;
    for (int p = 2; 1LL * p * p < MAX; p++) {
        if (is_prime[p]) {
            for (int i = p * p; i < MAX; i += p)
                is_prime[i] = false;
        }
    }
}

Số Học • Thừa Số O(sqrt(N))

Đếm Số Lượng Ước Số $d(N)$

Nếu $N = p_1^{e_1} \cdot p_2^{e_2} \dots p_k^{e_k}$ thì số ước là d(N) = (e1 + 1)(e2 + 1)...(ek + 1). 

int count_divisors(long long n) {
    int cnt = 0;
    for (long long i = 1; i * i <= n; i++) {
        if (n % i == 0) {
            cnt++; // Uoc i
            if (i * i != n) cnt++; // Uoc n / i
        }
    }
    return cnt;
}
Mảng • Tiền Xử Lý O(1) Truy Vấn

Mảng Cộng Dồn 1 Chiều (Prefix Sum 1D)

Công thức: Sum(L, R) = Pref[R] - Pref[L - 1] với mảng 1-based. 

vector<long long> pref(n + 1, 0);
for (int i = 1; i <= n; i++)
    pref[i] = pref[i - 1] + a[i];

// Truy van [L, R] trong O(1):
long long sum = pref[R] - pref[L - 1];
Mảng • 2 Chiều O(1) Truy Vấn

Mảng Cộng Dồn 2 Chiều (Prefix Sum 2D)

Tổng hình chữ nhật từ góc $(x_1, y_1)$ đến $(x_2, y_2)$ trong $O(1)$: 

// 1. Khoi tao:
P[i][j] = A[i][j] + P[i-1][j] + P[i][j-1] - P[i-1][j-1];

// 2. Truy van [x1, y1] den [x2, y2]:

    return true;

long long low = 1, high = 1e18, ans = -1;
while (low <= high) {
    long long mid = low + (high - low) / 2;
    if (check(mid)) {

int left = 0, max_len = 0;

for (int right = 0; right < n; right++) {
    cur_sum += a[right];
    while (cur_sum > s && left <= right) {
        cur_sum -= a[left];
        left++;
    }
    max_len = max(max_len, right - left + 1);
}

void backtrack(int i, long long cur_sum, int count) {
    if (i == n) {
        // Kiem tra phuong an & cap nhat ket qua
        if (cur_sum <= S) ans = max(ans, count);
        return;
    }
    // Nhanh 1: Khong chon a[i]
    backtrack(i + 1, cur_sum, count);
    // Nhanh 2: Co chon a[i]
    backtrack(i + 1, cur_sum + a[i], count + 1);
}
Đồ Thị • Duyệt O(V + E)

Biểu Diễn Đồ Thị & Duyệt BFS Hàng Đợi

vector<int> adj[MAX];
bool visited[MAX];

void bfs(int start_node) {
    queue<int> q;
    q.push(start_node);
    visited[start_node] = true;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : adj[u]) {
            if (!visited[v]) {
                visited[v] = true;
                q.push(v);
            }
        }
    }
}
