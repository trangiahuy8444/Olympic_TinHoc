SUMSEQ — Tính Tổng Dãy Số Lớn ($N \le 10^9$)

Tệp: SUMSEQ.INP / OUT

Cho ba số nguyên dương $A, B, D$ ($A \le B \le 10^9, D \le 10^5$). Dãy số cấp số cộng gồm các phần tử: $A, A + D, A + 2D, \dots \le B$. Tính tổng tất cả phần tử không vượt quá $B$.

SUMSEQ.INP: 3 15 4
SUMSEQ.OUT: 36
Thuật toán $O(1)$ tránh TLE & Tràn số: 
#include <iostream>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    long long A, B, D;
    if (!(cin >> A >> B >> D)) return 0;

    if (A > B) { cout << 0 << "\n"; return 0; }

    long long n = (B - A) / D + 1;
    long long last = A + (n - 1) * D;

    // Tinh n * (A + last) / 2 bang cach chia truoc de chong tran so
    long long ans;
    if (n % 2 == 0) ans = (n / 2) * (A + last);
    else ans = n * ((A + last) / 2);

    cout << ans << "\n";
    return 0;
}

Mảng Đếm Tần Số (Counting Array) & Kỹ Thuật Xử Lý Xâu

Bẻ gãy độ phức tạp từ $O(N^2)$ xuống $O(N)$ bằng mảng đánh dấu tần số trực tiếp; làm chủ thư viện <string> và bảng mã ASCII trong các bài toán chuẩn hóa xâu.

PAIRSUM — Đếm Cặp Tổng Bằng K ($N \le 10^5, A_i \le 10^6$)

PAIRSUM.INP / OUT

Cho mảng $N$ số nguyên không âm và số $K$. Đếm số cặp $(i, j)$ với $i < j$ sao cho $A_i + A_j = K$. 

#include <iostream>
#include <vector>
using namespace std;

const int MAX = 2000005;
int cnt[MAX] = {0};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, k;
    if (!(cin >> n >> k)) return 0;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    long long ans = 0;
    for (int i = 0; i < n; i++) {
        int target = k - a[i];
        if (target >= 0 && target < MAX) {
            ans += cnt[target]; // Da gap bao nhieu so thoa man truoc do
        }
        if (a[i] >= 0 && a[i] < MAX) {
            cnt[a[i]]++;
        }
    }

    cout << ans << "\n";
    return 0;
}
Bài 2 • Xử lý xâu

PALIN — Xâu Đối Xứng Nâng Cao (Bỏ qua ký tự đặc biệt)

PALIN.INP / OUT

Kiểm tra xem xâu có đối xứng không sau khi đã loại bỏ dấu cách, dấu câu và không phân biệt hoa thường. Ví dụ: "A man, a plan, a canal: Panama!" $\to$ YES. 

#include <iostream>
#include <string>
#include <cctype>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    string s;
    if (!getline(cin, s)) return 0;

    string clean = "";
    for (char c : s) {
        if (isalnum(c)) clean += tolower(c);
    }

    int left = 0, right = (int)clean.length() - 1;
    bool ok = true;
    while (left < right) {
        if (clean[left] != clean[right]) { ok = false; break; }
        left++; right--;
    }

    if (ok) cout << "YES\n";
    else cout << "NO\n";

    return 0;
}

Kỹ Thuật Tiền Xử Lý: Mảng Cộng Dồn (Prefix Sum 1D & 2D)

Tính tổng đoạn $[L, R]$ trong $O(1)$ thay vì duyệt lặp $O(N)$ bằng công thức $Pref[R] - Pref[L-1]$; ứng dụng xử lý hàng triệu truy vấn tổng chỉ trong chớp mắt.

Công Thức Cốt Lõi Prefix Sum

Khởi tạo (1-based index):

Pref[0] = 0 Pref[i] = Pref[i - 1] + A[i] (1 ≤ i ≤ N)

Truy vấn trong $O(1)$:

Sum(L, R) = Pref[R] - Pref[L - 1]

Nếu có $Q = 10^5$ truy vấn: Duyệt thông thường mất $O(N \cdot Q) = 10^{10}$ phép tính (chắc chắn dính TLE). Dùng Prefix Sum chỉ mất $O(N + Q) \approx 2 \cdot 10^5$ phép tính (chạy dưới 0.05 giây!).

Mở Rộng: Prefix Sum 2D Trên Bảng

Tính tổng hình chữ nhật từ $(x_1, y_1)$ đến $(x_2, y_2)$ trong $O(1)$:

Pref[x][y] = A[x][y] + Pref[x-1][y] + Pref[x][y-1] - Pref[x-1][y-1];

Sum = Pref[x2][y2] - Pref[x1-1][y2] - Pref[x2][y1-1] + Pref[x1-1][y1-1];

Mảng Hiệu (Difference Array) & Sắp Xếp Tùy Biến

Tăng giảm đoạn $[L, R]$ thêm $V$ đơn vị trong $O(1)$ thay vì $O(N)$; áp dụng giải quyết triệt để các bài toán đặt phòng, phục vụ quán ăn và chồng lấn thời gian.

Kỹ Thuật Mảng Hiệu (Difference Array)

Khi cần thực hiện $Q$ lần thao tác: "Cộng thêm $V$ vào tất cả phần tử từ chỉ số $L$ đến $R$":

// 1. Cap nhat trong O(1):
diff[L] += V;
diff[R + 1] -= V;
// 2. Khoi phuc mang sau Q lan cap nhat: O(N)
for (int i = 1; i <= n; i++) cur += diff[i]; a[i] = cur;

Tổng độ phức tạp là $O(Q + N)$ thay vì $O(Q \cdot N)$, cứu nguy ngoạn mục cho bài toán cập nhật đoạn.

Sắp Xếp Tùy Biến (Custom Comparator) 

struct ThiSinh {
    string ten;
    int diem;
};

// Tra ve true neu a dung truoc b
bool cmp(const ThiSinh &a, const ThiSinh &b) {
    if (a.diem != b.diem) return a.diem > b.diem; // Diem cao xep truoc
    return a.ten < b.ten; // Bang diem: xep theo ten Alphabet
}

sort(ds.begin(), ds.end(), cmp);

RESTAURANT — Quản Lý Đặt Bàn Nhà Hàng ($N \le 10^5$)

RESTAURANT.INP / OUT

Có $N$ đoàn khách đặt bàn, đoàn thứ $i$ ăn từ mốc $L_i$ đến $R_i$. Hỏi nhà hàng cần tối thiểu bao nhiêu bàn để phục vụ tất cả khách không bị trùng? 

#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAX_TIME = 1000005;
int diff_tbl[MAX_TIME] = {0};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n;
    if (!(cin >> n)) return 0;

    int max_r = 0;
    for (int i = 0; i < n; i++) {
        int l, r;
        cin >> l >> r;
        diff_tbl[l] += 1;
        diff_tbl[r + 1] -= 1;
        max_r = max(max_r, r);
    }

    int cur_tables = 0, ans = 0;
    for (int t = 1; t <= max_r; t++) {
        cur_tables += diff_tbl[t];
        ans = max(ans, cur_tables);
    }

    cout << ans << "\n";
    return 0;
}

Bảng Giới Hạn Dữ Liệu & Cẩm Nang Phòng Tránh Tràn Số (Overflow)

Tràn số nguyên là nguyên nhân số một khiến học sinh bị mất 50% - 100% số điểm bài toán dù tư duy thuật toán hoàn toàn chính xác. Khi giá trị vượt quá giới hạn lưu trữ của kiểu dữ liệu, bit dấu bị đảo ngược khiến kết quả trở thành một số âm vô lý!

Kiểu dữ liệu Kích thước Khoảng giá trị an toàn Ứng dụng thực tế trong thi đấu
int 32-bit (4 bytes) $-2 \cdot 10^9$ đến $+2 \cdot 10^9$ Biến đếm vòng lặp $i, j, n \le 10^9$, chỉ số mảng, giá trị phần tử nhỏ.
long long 64-bit (8 bytes) $-9 \cdot 10^{18}$ đến $+9 \cdot 10^{18}$ Tính tổng mảng, tích 2 số lớn, cấp số cộng, số học, đếm luỹ thừa.
unsigned long long 64-bit không dấu $0$ đến $1.8 \cdot 10^{19}$ Thuật toán Băm chuỗi (String Hashing) với cơ chế tự động tràn modulo $2^{64}$.
double / long double 64 / 80-bit Số thực chính xác cao Hình học tính toán, toạ độ thực. Chú ý: tránh dùng toán tử so sánh bằng == với số thực!

Ba Bẫy Tràn Số Kinh Điển Học Sinh Thường Mắc Phải:

Bẫy 1: Nhân 2 biến int

Biến a = 10^6, b = 10^6. Nếu viết:
long long c = a * b; // SAI!
C++ nhân a * b ở kiểu 32-bit trước khi gán $\to$ tràn số!

long long c = 1LL * a * b;
Bẫy 2: Công thức tổng cấp số cộng

Công thức $S = \frac{n(A + \text{last})}{2}$. Nếu lấy $n \times (A + \text{last})$ trước, tích có thể đạt $2 \cdot 10^{18} \to$ tràn long long!

if (n % 2 == 0) ans = (n/2)*(A+last);
else ans = n*((A+last)/2);
Bẫy 3: Khai báo mảng lớn trong main()

Bộ nhớ Stack cho hàm main() chỉ khoảng 1MB - 8MB. Viết int a[1000000]; trong hàm sẽ gây Segmentation Fault!

Khai báo mảng toàn cục (Global) hoặc dùng vector<int> a(n);

Quy Tắc Vàng: 1 Giây $\approx 10^8$ Phép Tính & Bản Đồ Chọn Thuật Toán

Quy tắc sống còn của mọi lập trình viên thi đấu: Máy chấm Olympic xử lý được khoảng $10^8$ phép toán cơ bản trong 1.0 giây. Nhìn vào giới hạn $N$ trong đề bài, bạn có thể biết trước chính xác thuật toán cần dùng mà không cần đoán mò!

Giới hạn $N$ trong đề bài Độ phức tạp mục tiêu Các thuật toán chuẩn tương ứng Đánh giá khả năng đạt điểm
$N \le 20$ $O(2^N), O(N!)$ Quay lui (Backtracking), vét cạn tập con, Bitmask DP. Thường là Subtask 1 để ăn điểm khởi đầu.
$N \le 500$ $O(N^3)$ 3 vòng lặp for lồng nhau, thuật toán Floyd-Warshall tìm đường đi ngắn nhất. Subtask 1 - 2 các bài đồ thị hoặc dãy con.
$N \le 2000 \dots 5000$ $O(N^2)$ 2 vòng for lồng nhau, Quy hoạch động bảng 2 chiều, Dijkstra thường. Subtask 2 (khoảng 40% - 60% số điểm bài toán).
$N \le 10^5 \dots 2 \cdot 10^5$ $O(N \log N)$ Sắp xếp (std::sort), Tìm kiếm nhị phân (Binary Search), Segment Tree, Fenwick Tree (BIT). Chuẩn ăn trọn vẹn 100% điểm của bài phân loại!
$N \le 10^6 \dots 10^7$ $O(N)$ Mảng cộng dồn (Prefix Sum), Mảng đếm tần số, Kỹ thuật 2 con trỏ (Two Pointers), Cửa sổ trượt (Sliding Window). Thuật toán tuyến tính tối ưu tuyệt đối!
$N \ge 10^9 \dots 10^{18}$ $O(1) \text{ hoặc } O(\log N)$ Công thức Toán học đóng, Cấp số cộng/nhân, Luỹ thừa nhị phân (Binary Exponentiation), Ước số chung lớn nhất (GCD). Chạy tức thì dưới 1 mili-giây!

Hai Thuật Toán Mẫu Kinh Điển Buổi 1 (Chống TLE & Tràn Số)

Toán Học $O(1)$

SUMSEQ — Tổng Dãy Cấp Số Cộng ($N \le 10^9$)

Công thức số số hạng $n = \lfloor(B - A)/D\rfloor + 1$. Tổng $S = \frac{n \cdot (A + \text{last})}{2}$. Kỹ thuật chia 2 trước cho số chẵn ngăn chặn triệt để tràn số $9 \cdot 10^{18}$. 

long long n = (B - A) / D + 1;
long long last = A + (n - 1) * D;
long long ans;
if (n % 2 == 0) ans = (n / 2) * (A + last);
else ans = n * ((A + last) / 2);
cout << ans << "
";
Sắp Xếp $O(N \log N)$

PAIRPROD — Tích 2 Số Lớn Nhất ($A_i \le 10^9$)

Khi $A_i \le 10^9$, tích 2 số có thể đạt tới $10^{18}$. Cần sắp xếp mảng tăng dần rồi lấy 2 phần tử cuối mảng, nhân bằng kiểu long long. 

sort(a.begin(), a.end());
// 2 phần tử lớn nhất nằm ở cuối mảng đã sắp xếp
long long ans = 1LL * a[n - 1] * a[n - 2];
cout << ans << "
";

Phép Tính Modulo & Lũy Thừa Nhị Phân

Cộng: (a + b) % MOD
Trừ: (a - b % MOD + MOD) % MOD
Nhân: (1LL * (a % MOD) * (b % MOD)) % MOD
long long binpow(long long a, long long b, long long m) {
    long long res = 1; a %= m;
    while (b > 0) {
        if (b & 1) res = (1LL * res * a) % m;
        a = (1LL * a * a) % m;
        b >>= 1;
    }
    return res;
}

Hàm có sẵn: __gcd(a, b) trong <algorithm>. Công thức liên hệ: a * b = gcd(a, b) * lcm(a, b). 

long long gcd(long long a, long long b) {
    return b == 0 ? a : gcd(b, a % b);

long long lcm(long long a, long long b) {
    // Chia truoc de chong tran so a * b!
    return (a / __gcd(a, b)) * b;

void bfs(int start_node) {
    queue<int> q;
    q.push(start_node);
    visited[start_node] = true;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : adj[u]) {
            if (!visited[v]) {
                visited[v] = true;
                q.push(v);
            }
        }
    }
}
