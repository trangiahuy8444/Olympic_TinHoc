\section*{\color{AsianDarkBlue}I. TIẾN TRÌNH LÊN LỚP CHI TIẾT (90 PHÚT)}
\begin{itemize}
    \setlength{\itemsep}{2pt}
    \item \textbf{[00:00 -- 00:15] Khởi động, Fast-Track Diagnostic \& Setup Code::Blocks (15 phút):}
    \begin{itemize}
        \item Chiếu 3 câu hỏi chẩn đoán nhanh: (1) \texttt{5 / 2 = ?}; (2) \texttt{int} có chứa được 10 tỷ không?; (3) $1 \le x \le 10$ viết thế nào trong C++?
        \item \textbf{Chiến lược dạy nhanh:} Nếu học sinh đã rõ, lướt qua cú pháp cơ bản (tiết kiệm 20 phút), dồn thời lượng cho bẫy tràn số và bảng Dry Run!
        \item Cấu hình Code::Blocks (\texttt{-std=c++17}, \texttt{-O2}) và 3 phím tắt vàng: \texttt{Ctrl+Shift+N}, \texttt{Ctrl+S}, \texttt{F9}.
    \end{itemize}
    \item \textbf{[00:15 -- 00:45] Chuyên đề 1: Bản chất Biến nhớ, Kiểu dữ liệu \& Bẫy tràn số (30 phút):}
    \begin{itemize}
        \item Ẩn dụ biến là chiếc hộp có nhãn; bẫy giá trị rác nếu không khởi tạo; phân biệt \texttt{int} ($2 \cdot 10^9$) và \texttt{long long} ($9 \cdot 10^{18}$).
        \item Live-code chứng minh bẫy tràn số khi nhân: \texttt{long long s = 1LL * a * b;}.
    \end{itemize}
    \item \textbf{[00:45 -- 01:15] Chuyên đề 2: Toán tử số học, Rẽ nhánh if/else \& Vòng lặp for (30 phút):}
    \begin{itemize}
        \item Phân biệt chia nguyên \texttt{5 / 2 = 2} và chia dư \texttt{5 \% 2 = 1}; ứng dụng kiểm tra chẵn lẻ và tách chữ số hàng đơn vị.
        \item Rẽ nhánh \texttt{if/else}, bẫy toán học: viết \texttt{1 <= x \&\& x <= 10} thay vì \texttt{1 <= x <= 10}.
        \item Kẻ bảng Dry Run chạy từng bước bằng tay tính tổng $S = 1 + 2 + \dots + n$ bằng vòng lặp \texttt{for}.
    \end{itemize}
    \item \textbf{[01:15 -- 01:30] Thực hành tại lớp, Giao Cẩm nang Tự học A--Z \& BTVN (15 phút):}
    \begin{itemize}
        \item Hướng dẫn mở tệp \texttt{freopen} đọc file \texttt{.INP} và xuất \texttt{.OUT}.
        \item Giao nhiệm vụ tự học: Tự đọc và gõ lại các bài trong \textbf{Cẩm nang Tự học C++ Tập 01}; hoàn thành Bộ bài tập tăng cường cú pháp Tuần 01.
    \end{itemize}
\end{itemize}

\subsection*{\color{AsianBlue}2. Minh họa bẫy tràn số nguyên (Integer Overflow) \& Cách khắc phục}
\begin{lstlisting}[language=C++]
int a = 1000000, b = 1000000;

// Bay tran so: a * b tinh trong kieu int 32-bit (vuot 2 ty -> ra so am)
long long sai = a * b;          

// Cach xu ly chuan: Dung 1LL ep phep nhan len kieu 64-bit an toan
long long dung = 1LL * a * b;   
\end{lstlisting}

\subsection*{\color{AsianBlue}3. Bảng Trace Dry Run: Tính tổng $S = 1 + 2 + 3 + 4$ (Vòng lặp for)}
\begin{center}
\footnotesize
\begin{tabular}{@{}ccccccc@{}}
\toprule
\textbf{Vòng} & \textbf{Biến $i$} & \textbf{Điều kiện $i \le 4$} & \textbf{$S$ trước cộng} & \textbf{Lệnh $S = S + i$} & \textbf{$S$ sau cộng} & \textbf{Bước nhảy $i$++} \\
\midrule
Khởi tạo & - & - & - & $S = 0$ & 0 & $i = 1$ \\
1 & 1 & $1 \le 4$ (\textbf{Đúng}) & 0 & $S = 0 + 1$ & \textbf{1} & $i$ tăng lên 2 \\
2 & 2 & $2 \le 4$ (\textbf{Đúng}) & 1 & $S = 1 + 2$ & \textbf{3} & $i$ tăng lên 3 \\
3 & 3 & $3 \le 4$ (\textbf{Đúng}) & 3 & $S = 3 + 3$ & \textbf{6} & $i$ tăng lên 4 \\
4 & 4 & $4 \le 4$ (\textbf{Đúng}) & 6 & $S = 6 + 4$ & \textbf{10} & $i$ tăng lên 5 \\
Thoát & 5 & $5 \le 4$ (\textbf{Sai!}) & 10 & \textit{Không chạy} & \textbf{10} & Dừng vòng lặp \\
\bottomrule
\end{tabular}
\end{center}
