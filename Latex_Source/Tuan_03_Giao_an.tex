    \item \textbf{Bước ngoặt thuật toán:} Làm chủ kỹ thuật \textbf{Mảng đếm tần số (Counting Array / Mảng đánh dấu)}, chuyển đổi độ phức tạp từ $O(N^2)$ (TLE) về $O(N)$ tức thì.
    \item \textbf{Kỹ năng Olympic:} Xử lý kỹ thuật dời trục (\textit{Offset}) khi dãy có số âm, tiếp cận tư duy kinh điển "Đổi bộ nhớ lấy thời gian" (\textit{Space-Time Tradeoff}).
\end{enumerate}
\end{tcolorbox}

\section*{\color{AsianDarkBlue}I. TIẾN TRÌNH LÊN LỚP CHI TIẾT (90 PHÚT)}
\begin{itemize}
    \setlength{\itemsep}{2pt}
    \item \textbf{[00:00 -- 00:15] Kiểm tra bài cũ \& Fast-Track Diagnostic Tuần 3 (15 phút):}
    \begin{itemize}
        \item Chiếu 3 câu hỏi chẩn đoán nhanh: (1) Hàm \texttt{void} khác hàm có kiểu trả về (\texttt{int}, \texttt{bool}) ở điểm mấu chốt nào? (Không có giá trị trả về, không cần \texttt{return}); (2) Cú pháp \texttt{void tang(int \&x)} có dấu \texttt{\&} khác gì \texttt{void tang(int x)}? (Có \texttt{\&} là tham chiếu làm đổi biến gốc; không có là tham trị chỉ đổi bản sao); (3) Để tìm số xuất hiện nhiều nhất trong $O(N)$ thay vì $O(N^2)$, ta dùng cấu trúc gì? (Mảng đếm tần số \texttt{cnt[x]++}).
        \item \textbf{Chiến lược dạy nhanh:} Nếu học sinh đã rõ, lướt nhanh phần lý thuyết cú pháp hàm, dồn toàn lực vào phân tích bài toán thực chiến mảng đếm tần số và kỹ thuật dời trục Offset số âm!
        \item Code Audit BTVN Tuần 2: Thói quen khai báo mảng toàn cục ngoài \texttt{main()}, bẫy khởi tạo Min/Max số âm và lỗi tràn biên \textit{out-of-bounds}.
    \end{itemize}
    \item \textbf{[00:15 -- 00:40] Chuyên đề 1: Hàm \& Truyền tham chiếu \texttt{\&} (25 phút):}
    \begin{itemize}
        \item Vẽ mô hình ô nhớ RAM trên bảng để phân tích bản sao tham trị vs trỏ thẳng ô nhớ tham chiếu; học sinh code thử nghiệm hàm \texttt{my\_swap(int \&u, int \&v)}.
    \end{itemize}
    \item \textbf{[00:40 -- 01:10] Chuyên đề 2: Đột phá Mảng đếm tần số $O(N)$ (30 phút):}
    \begin{itemize}
        \item Trực quan hóa so sánh: 2 vòng lặp for lồng nhau đơ máy ($10^{10}$ phép tính) vs mảng \texttt{cnt[x]++} chạy trong $0.01$s; kỹ thuật dời trục \textit{Offset} khi dãy có số âm.
    \end{itemize}

\subsection*{\color{AsianBlue}1. Phân biệt Tham trị vs. Tham chiếu (\texttt{\&})}
\begin{lstlisting}[language=C++]
// Tham tri: tao ban sao moi tren Stack, khong doi bien goc
void tang_mot(int x) { x++; }

// Tham chieu (&): nhan truc tiep dia chi o nho, bien goc thay doi
void tang_mot_that(int &x) { x++; }

// Ham hoan doi 2 so dung tham chieu
void my_swap(long long &u, long long &v) {
    long long temp = u; u = v; v = temp;
}
\end{lstlisting}

\subsection*{\color{AsianBlue}2. Khuôn mẫu Mảng đếm tần số $O(N)$ chuẩn phòng thi}
\begin{lstlisting}[language=C++]
#include <iostream>
using namespace std;
const int MAX_VAL = 1000005;
int cnt[MAX_VAL]; // Mang dem toan cuc tren vung BSS (tu dong = 0)

int main() {
    freopen("UNIQUE.INP", "r", stdin);
    freopen("UNIQUE.OUT", "w", stdout);
    ios_base::sync_with_stdio(false); cin.tie(NULL);

    int n; if (!(cin >> n)) return 0;
    int max_val = 0;
    for (int i = 1; i <= n; i++) {
        int x; cin >> x;
        cnt[x]++; // Chi so la gia tri, noi dung la so lan xuat hien
        if (x > max_val) max_val = x;
    }
    // Dem so luong gia tri phan biet trong day
    int dem_phan_biet = 0;
    for (int v = 0; v <= max_val; v++) {
        if (cnt[v] > 0) dem_phan_biet++;
    }
    cout << dem_phan_biet << "\n";
    return 0;
}
\end{lstlisting}
